import os
import subprocess
import json
from datetime import datetime
from pathlib import Path

def analyze_codebase():
    """Analyze code quality metrics for the current project."""
    
    report = {
        "timestamp": datetime.now().isoformat(),
        "project": "trinity-hdm",
        "analysis_date": datetime.now().strftime("%Y-%m-%d"),
        "metrics": {},
        "issues": [],
        "recommendations": []
    }
    
    # Get project structure
    try:
        # Count files by type
        file_counts = {}
        total_lines = 0
        total_files = 0
        
        for root, dirs, files in os.walk("."):
            # Skip hidden directories and virtual environments
            dirs[:] = [d for d in dirs if not d.startswith('.') and d not in ['__pycache__', 'venv', 'env', 'node_modules']]
            
            for file in files:
                if file.startswith('.'):
                    continue
                    
                filepath = os.path.join(root, file)
                ext = os.path.splitext(file)[1]
                
                if ext:
                    file_counts[ext] = file_counts.get(ext, 0) + 1
                
                # Count lines for text files
                if ext in ['.py', '.js', '.ts', '.md', '.txt', '.json', '.yml', '.yaml', '.html', '.css']:
                    try:
                        with open(filepath, 'r', encoding='utf-8', errors='ignore') as f:
                            lines = len(f.readlines())
                            total_lines += lines
                            total_files += 1
                    except:
                        pass
        
        report["file_statistics"] = {
            "total_files_analyzed": total_files,
            "total_lines_of_code": total_lines,
            "file_types": file_counts
        }
        
    except Exception as e:
        report["errors"] = [f"File analysis error: {str(e)}"]
    
    # Check for common code quality issues
    issues = []
    
    # 1. Check for large files (> 500 lines)
    large_files = []
    for root, dirs, files in os.walk("."):
        dirs[:] = [d for d in dirs if not d.startswith('.')]
        for file in files:
            if file.endswith('.py'):
                filepath = os.path.join(root, file)
                try:
                    with open(filepath, 'r', encoding='utf-8') as f:
                        lines = len(f.readlines())
                        if lines > 500:
                            large_files.append({
                                "file": filepath[2:],  # Remove './' prefix
                                "lines": lines,
                                "issue": "File exceeds 500 lines (consider refactoring)"
                            })
                except:
                    pass
    
    if large_files:
        issues.append({
            "category": "Maintainability",
            "severity": "medium",
            "description": "Large files detected",
            "details": large_files
        })
    
    # 2. Check for TODO/FIXME comments
    todo_count = 0
    todo_files = {}
    
    for root, dirs, files in os.walk("."):
        dirs[:] = [d for d in dirs if not d.startswith('.')]
        for file in files:
            if file.endswith(('.py', '.js', '.ts', '.md')):
                filepath = os.path.join(root, file)
                try:
                    with open(filepath, 'r', encoding='utf-8') as f:
                        content = f.read().lower()
                        if 'todo' in content or 'fixme' in content:
                            todo_count += 1
                            todo_files[filepath[2:]] = "Contains TODO/FIXME"
                except:
                    pass
    
    if todo_count > 0:
        issues.append({
            "category": "Code Cleanliness",
            "severity": "low",
            "description": f"Found {todo_count} files with TODO/FIXME comments",
            "details": todo_files
        })
    
    # 3. Check documentation coverage
    doc_files = []
    for root, dirs, files in os.walk("."):
        dirs[:] = [d for d in dirs if not d.startswith('.')]
        for file in files:
            if file.endswith('.md'):
                doc_files.append(os.path.join(root, file)[2:])
    
    report["documentation"] = {
        "markdown_files": doc_files,
        "count": len(doc_files)
    }
    
    # Calculate basic metrics
    if total_files > 0:
        avg_lines_per_file = total_lines / total_files
        report["metrics"]["average_lines_per_file"] = round(avg_lines_per_file, 2)
        
        # Complexity estimation (simplified)
        if avg_lines_per_file > 200:
            complexity = "high"
        elif avg_lines_per_file > 100:
            complexity = "medium"
        else:
            complexity = "low"
        
        report["metrics"]["estimated_complexity"] = complexity
    
    report["issues"] = issues
    
    # Generate recommendations
    recommendations = []
    
    if issues:
        recommendations.append("Address large files by refactoring into smaller modules")
        recommendations.append("Resolve TODO/FIXME comments or create issues for them")
    
    if len(doc_files) < 5 and total_files > 20:
        recommendations.append("Improve documentation coverage - add more README files")
    
    recommendations.append("Implement automated testing if not already present")
    recommendations.append("Consider adding type hints for better code clarity")
    recommendations.append("Regularly run code quality tools (pylint, black, mypy)")
    
    report["recommendations"] = recommendations
    
    # Overall assessment
    issue_count = len(issues)
    if issue_count == 0:
        assessment = "EXCELLENT"
    elif issue_count <= 2:
        assessment = "GOOD"
    elif issue_count <= 5:
        assessment = "FAIR"
    else:
        assessment = "NEEDS_IMPROVEMENT"
    
    report["overall_assessment"] = {
        "score": assessment,
        "issues_found": issue_count,
        "next_review": (datetime.now() + timedelta(days=7)).strftime("%Y-%m-%d")
    }
    
    return report
