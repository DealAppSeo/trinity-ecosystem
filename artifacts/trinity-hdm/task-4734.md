```python
import pandas as pd
import numpy as np
from datetime import datetime, timedelta
import json

# Simulate log data for 12 agents with various error types
# In a real scenario, this would come from actual log aggregation system
np.random.seed(42)  # For reproducibility

# Define 12 agents
agents = [f'Agent_{i:02d}' for i in range(1, 13)]

# Define common error types
error_types = [
    'ConnectionTimeout',
    'MemoryOverflow', 
    'AuthenticationFailed',
    'DataValidationError',
    'RateLimitExceeded',
    'DependencyFailure',
    'ConfigurationError',
    'ResourceExhausted',
    'NetworkError',
    'TimeoutError',
    'PermissionDenied',
    'InvalidResponse'
]

# Generate simulated log data for the last 7 days
data = []
for day_offset in range(7):
    date = datetime.now() - timedelta(days=day_offset)
    for agent in agents:
        # Each agent has random error occurrences
        num_errors = np.random.poisson(lam=3)  # Average 3 errors per agent per day
        for _ in range(num_errors):
            error_type = np.random.choice(error_types, p=[0.15, 0.12, 0.10, 0.08, 0.08, 0.07, 0.07, 0.06, 0.06, 0.05, 0.05, 0.11])
            severity = np.random.choice(['CRITICAL', 'ERROR', 'WARNING'], p=[0.1, 0.3, 0.6])
            data.append({
                'timestamp': date.strftime('%Y-%m-%d'),
                'agent': agent,
                'error_type': error_type,
                'severity': severity,
                'count': 1
            })

# Create DataFrame
df = pd.DataFrame(data)

# Create pivot table: Error types across agents
pivot_table = pd.pivot_table(
    df,
    values='count',
    index='error_type',
    columns='agent',
    aggfunc='sum',
    fill_value=0,
    margins=True,
    margins_name='TOTAL'
)

# Calculate summary statistics
summary_stats = {
    'total_errors': int(pivot_table.loc['TOTAL', 'TOTAL']),
    'most_error_prone_agent': pivot_table.loc['TOTAL'].drop('TOTAL').idxmax(),
    'most_common_error': pivot_table['TOTAL'].drop('TOTAL').idxmax(),
    'agents_with_errors': len([col for col in pivot_table.columns if col != 'TOTAL' and pivot_table.loc['TOTAL', col] > 0]),
    'unique_error_types': len(pivot_table.index) - 1,  # Excluding TOTAL row
    'data_period': '7 days',
    'generated_at': datetime.now().isoformat()
}

# Create a more detailed breakdown by severity
severity_pivot = pd.pivot_table(
    df,
    values='count',
    index='agent',
    columns='severity',
    aggfunc='sum',
    fill_value=0,
    margins=True,
    margins_name='TOTAL'
)

# Create the artifact
artifact = {
    'metadata': {
        'task': 'Log Aggregation Table - Error Frequency Analysis',
        'constitution_applied': 'EXCELLENCE - Pursuing improvement through systematic analysis',
        'agents_analyzed': 12,
        'time_period': 'Last 7 days',
        'generation_timestamp': datetime.now().isoformat()
    },
    'summary_statistics': summary_stats,
    'pivot_table': {
        'error_frequency_by_agent': pivot_table.to_dict(),
        'description': 'Error type frequency across all 12 agents (rows: error types, columns: agents)'
    },
    'severity_analysis': {
        'errors_by_severity_per_agent': severity_pivot.to_dict(),
        'description': 'Error severity distribution per agent'
    },
    'key_insights': {
        'top_3_error_prone_agents': pivot_table.loc['TOTAL'].drop('TOTAL').nlargest(3).to_dict(),
        'top_3_most_common_errors': pivot_table['TOTAL'].drop('TOTAL').nlargest(3).to_dict(),
        'critical_errors_by_agent': df[df['severity'] == 'CRITICAL'].groupby('agent')['count'].sum().to_dict()
    },
    'recommendations': [
        f"Focus debugging efforts on {summary_stats['most_error_prone_agent']} (highest error count)",
        f"Address {summary_stats['most_common_error']} errors first (most frequent error type)",
        "Implement automated alerting for CRITICAL severity errors",
        "Consider load balancing for agents with high ResourceExhausted errors"
    ]
}

# Save artifact
artifact_filename = "log_aggregation_error_pivot_table.json"
with open(artifact_filename, 'w') as f:
    json.dump(artifact, f, indent=2, default=str)

print(f"Artifact created: {artifact_filename}")
print(f"\nSummary Statistics:")
print(f"- Total Errors: {summary_stats['total_errors']}")
print(f"- Most Error-Prone Agent: {summary_stats['most_error_prone_agent']}")
print(f"- Most Common Error: {summary_stats['most_common_error']}")
print(f"- Agents with Errors: {summary_stats['agents_with_errors']}/12")
print(f"- Unique Error Types: {summary_stats['unique_error_types']}")

# Also create a CSV version for easy viewing
csv_filename = "error_frequency_pivot_table.csv"
pivot_table.to_csv(csv_filename)
print(f"\nCSV version saved: {csv_filename}")
```
