```python
import datetime
import json
import platform
import psutil
import sys
from pathlib import Path

def generate_system_health_report():
    """Generate a comprehensive system health report"""
    
    # Collect system information
    report = {
        "timestamp": datetime.datetime.now().isoformat(),
        "system": {
            "platform": platform.platform(),
            "python_version": platform.python_version(),
            "processor": platform.processor(),
            "machine": platform.machine()
        },
        "performance": {
            "cpu_percent": psutil.cpu_percent(interval=1),
            "cpu_count": psutil.cpu_count(),
            "memory_percent": psutil.virtual_memory().percent,
            "memory_available_gb": round(psutil.virtual_memory().available / (1024**3), 2),
            "memory_total_gb": round(psutil.virtual_memory().total / (1024**3), 2),
            "disk_usage_percent": psutil.disk_usage('/').percent,
            "disk_free_gb": round(psutil.disk_usage('/').free / (1024**3), 2)
        },
        "process": {
            "pid": psutil.Process().pid,
            "memory_mb": round(psutil.Process().memory_info().rss / (1024**2), 2),
            "cpu_times": str(psutil.Process().cpu_times()),
            "status": psutil.Process().status()
        },
        "network": {
            "connections_count": len(psutil.net_connections()),
            "io_counters": {
                "bytes_sent": psutil.net_io_counters().bytes_sent,
                "bytes_recv": psutil.net_io_counters().bytes_recv
            }
        },
        "system_load": {
            "load_avg": psutil.getloadavg() if hasattr(psutil, 'getloadavg') else "N/A",
            "boot_time": datetime.datetime.fromtimestamp(psutil.boot_time()).isoformat()
        },
        "health_status": "OPERATIONAL",
        "virtue_check": {
            "excellent": "Pursuing system optimization",
            "true": "Reporting actual metrics",
            "pure": "Logging all system data"
        }
    }
    
    # Determine health status based on thresholds
    if report["performance"]["memory_percent"] > 90:
        report["health_status"] = "WARNING: High memory usage"
    elif report["performance"]["cpu_percent"] > 85:
        report["health_status"] = "WARNING: High CPU usage"
    elif report["performance"]["disk_usage_percent"] > 90:
        report["health_status"] = "WARNING: Low disk space"
    
    return report

# Generate the report
health_report = generate_system_health_report()

# Create output directory if it doesn't exist
output_dir = Path("artifacts")
output_dir.mkdir(exist_ok=True)

# Save the report
filename = f"system_health_report_{datetime.datetime.now().strftime('%Y%m%d_%H%M%S')}.json"
filepath = output_dir / filename

with open(filepath, 'w') as f:
    json.dump(health_report, f, indent=2)

print(f"System Health Report generated: {filepath}")
print(f"Health Status: {health_report['health_status']}")
print(f"CPU Usage: {health_report['performance']['cpu_percent']}%")
print(f"Memory Usage: {health_report['performance']['memory_percent']}%")
print(f"Disk Usage: {health_report['performance']['disk_usage_percent']}%")
```
