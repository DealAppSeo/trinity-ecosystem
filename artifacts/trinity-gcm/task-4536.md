import os
import ast
import subprocess
import json
from datetime import datetime
from pathlib import Path

def analyze_codebase():
    """Analyze code quality metrics for the current project."""
    
    print("🔍 Analyzing codebase structure...")
    
    # Get current directory structure
    base_path = Path(".")
    python_files = list(base_path.rglob("*.py"))
    md_files = list(base_path.rglob("*.md"))
    other_files = list(base_path.rglob("*"))
    
    # Basic metrics
    total_lines = 0
    total_functions = 0
    total_classes = 0
    imports = set()
    file_complexities = []
    
    # Analyze each Python file
    for py_file in python_files:
        try:
            with open(py_file, 'r', encoding='utf-8') as f:
                content = f.read()
                lines = content.split('\n')
                total_lines += len(lines)
                
                # Parse AST for deeper analysis
                tree = ast.parse(content)
                
                # Count functions and classes
                functions = [node for node in ast.walk(tree) if isinstance(node, ast.FunctionDef)]
                classes = [node for node in ast.walk(tree) if isinstance(node, ast.ClassDef)]
                total_functions += len(functions)
                total_classes += len(classes)
                
                # Collect imports
                for node in ast.walk(tree):
                    if isinstance(node, ast.Import):
                        for alias in node.names:
                            imports.add(alias.name)
                    elif isinstance(node, ast.ImportFrom):
                        if node.module:
                            imports.add(node.module)
                
                # Simple complexity metric (functions per file)
                if len(lines) > 0:
                    complexity = len(functions) / max(1, len(lines)/100)
                    file_complexities.append({
                        'file': str(py_file),
                        'lines': len(lines),
                        'functions': len(functions),
                        'classes': len(classes),
                        'complexity': round(complexity, 2)
                    })
                    
        except Exception as e:
            print(f"⚠️ Could not analyze {py_file}: {e}")
    
    # Check for common code quality issues
    issues = []
    
    # Check for large files
    for file_info in file_complexities:
        if file_info['lines'] > 300:
            issues.append(f"Large file: {file_info['file']} ({file_info['lines']} lines)")
        if file_info['complexity'] > 5:
            issues.append(f"High complexity: {file_info['file']} (score: {file_info['complexity']})")
    
    # Check documentation coverage
    doc_files_count = len(md_files)
    py_files_count = len(python_files)
    doc_coverage = (doc_files_count / max(1, py_files_count)) * 100
    
    # Generate report
    report = {
        "timestamp": datetime.now().isoformat(),
        "metrics": {
            "total_files": len(other_files),
            "python_files": py_files_count,
            "documentation_files": doc_files_count,
            "total_lines_of_code": total_lines,
            "total_functions": total_functions,
            "total_classes": total_classes,
            "unique_imports": len(imports),
            "average_functions_per_file": round(total_functions / max(1, py_files_count), 2),
            "documentation_coverage": round(doc_coverage, 1),
            "average_file_size": round(total_lines / max(1, py_files_count), 2)
        },
        "top_imports": sorted(list(imports))[:10],
        "file_complexities": sorted(file_complexities, key=lambda x: x['complexity'], reverse=True)[:5],
        "issues_found": issues,
        "recommendations": []
    }
    
    # Add recommendations based on analysis
    if doc_coverage < 50:
        report["recommendations"].append("Increase documentation coverage (currently {doc_coverage}%)")
    
    if len(issues) > 5:
        report["recommendations"].append("Refactor large/complex files to improve maintainability")
    
    if py_files_count > 0 and total_functions / py_files_count > 10:
        report["recommendations"].append("Consider breaking down files with high function density")
    
    return report

# Execute analysis
