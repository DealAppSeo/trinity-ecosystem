def calculate_performance_scores(metrics):
    """Calculate performance scores for each table"""
    
    scores = {}
    
    for table, data in metrics.items():
        # Weighted scoring formula
        read_score = (1000 / max(data['avg_read_latency_ms'], 0.1)) * 0.3
        write_score = (1000 / max(data['avg_write_latency_ms'], 0.1)) * 0.3
        throughput_score = ((data['read_throughput_ops_sec'] + data['write_throughput_ops_sec']) / 100) * 0.2
        cache_score = data['cache_hit_rate_percent'] * 0.1
        index_score = data['index_utilization_percent'] * 0.1
        
        total_score = read_score + write_score + throughput_score + cache_score + index_score
        normalized_score = min(100, total_score / 10)  # Normalize to 0-100 scale
        
        scores[table] = {
            "performance_score": round(normalized_score, 1),
            "read_score": round(read_score, 1),
            "write_score": round(write_score, 1),
            "throughput_score": round(throughput_score, 1)
        }
    
    return scores
