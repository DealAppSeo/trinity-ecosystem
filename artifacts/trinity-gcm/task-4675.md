def analyze_python_file(filepath):
    """Analyze a single Python file for quality metrics"""
    metrics = {
        "filename": str(filepath.name),
        "lines": 0,
        "functions": 0,
        "classes": 0,
        "complexity": 0,
        "issues": []
    }
    
    try:
        with open(filepath, 'r', encoding='utf-8') as f:
            content = f.read()
            metrics["lines"] = len(content.split('\n'))
            
        # Parse AST for basic analysis
        tree = ast.parse(content)
        
        # Count functions and classes
        metrics["functions"] = len([node for node in ast.walk(tree) if isinstance(node, ast.FunctionDef)])
        metrics["classes"] = len([node for node in ast.walk(tree) if isinstance(node, ast.ClassDef)])
        
        # Simple complexity metric (functions + classes per 100 lines)
        if metrics["lines"] > 0:
            metrics["complexity"] = (metrics["functions"] + metrics["classes"]) / (metrics["lines"] / 100)
        
        # Check for common issues
        issues = check_python_issues(tree, content)
        metrics["issues"] = issues
        
    except Exception as e:
        metrics["issues"].append(f"Parse error: {str(e)}")
    
    return metrics

def check_python_issues(tree, content):
    """Check for common Python code issues"""
    issues = []
    
    # Check for print statements (potential debug code)
    for node in ast.walk(tree):
        if isinstance(node, ast.Call):
            if isinstance(node.func, ast.Name) and node.func.id == 'print':
                issues.append("Found print statement - consider using logging")
    
    # Check for broad exceptions
    for node in ast.walk(tree):
        if isinstance(node, ast.ExceptHandler):
            if node.type is None:  # Bare except
                issues.append("Found bare except clause - specify exception type")
    
    # Check line length
    lines = content.split('\n')
    for i, line in enumerate(lines, 1):
        if len(line) > 100:  # PEP 8 recommends 79, using 100 as threshold
            issues.append(f"Line {i}: Exceeds 100 characters ({len(line)} chars)")
    
    return issues
