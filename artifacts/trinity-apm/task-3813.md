```python
import time
import logging
from datetime import datetime, timedelta
from functools import wraps
from typing import Dict, Any, Optional
import statistics

logging.basicConfig(level=logging.INFO)
logger = logging.getLogger(__name__)

class FailureMetrics:
    """Track and analyze failure patterns"""
    
    def __init__(self, window_hours: int = 24):
        self.failures = []
        self.successes = []
        self.window_hours = window_hours
        
    def record_failure(self, task_name: str, error: str, duration: float):
        entry = {
            'timestamp': datetime.now(),
            'task': task_name,
            'error': error,
            'duration': duration
        }
        self.failures.append(entry)
        self._clean_old_entries()
        
    def record_success(self, task_name: str, duration: float):
        entry = {
            'timestamp': datetime.now(),
            'task': task_name,
            'duration': duration
        }
        self.successes.append(entry)
        self._clean_old_entries()
        
    def _clean_old_entries(self):
        cutoff = datetime.now() - timedelta(hours=self.window_hours)
        self.failures = [f for f in self.failures if f['timestamp'] > cutoff]
        self.successes = [s for s in self.successes if s['timestamp'] > cutoff]
        
    def failure_rate(self, task_name: Optional[str] = None) -> float:
        """Calculate failure rate for task or overall"""
        relevant_failures = self.failures
        relevant_successes = self.successes
        
        if task_name:
            relevant_failures = [f for f in self.failures if f['task'] == task_name]
            relevant_successes = [s for s in self.successes if s['task'] == task_name]
            
        total = len(relevant_failures) + len(relevant_successes)
        if total == 0:
            return 0.0
        return len(relevant_failures) / total
        
    def get_patterns(self) -> Dict[str, Any]:
        """Identify failure patterns"""
        if not self.failures:
            return {}
            
        # Group by error type
        error_groups = {}
        for failure in self.failures:
            error_type = failure['error']
            if error_type not in error_groups:
                error_groups[error_type] = []
            error_groups[error_type].append(failure)
            
        # Analyze temporal patterns
        failure_times = [f['timestamp'].hour for f in self.failures]
        hour_distribution = {}
        for hour in range(24):
            hour_distribution[hour] = failure_times.count(hour)
            
        return {
            'total_failures': len(self.failures),

class CircuitBreaker:
    """Circuit breaker pattern implementation"""
    
    def __init__(self, name: str, failure_threshold: int = 5, 
                 reset_timeout: int = 60):
        self.name = name
        self.failure_threshold = failure_threshold
        self.reset_timeout = reset_timeout
        self.failure_count = 0
        self.last_failure_time = None
        self.state = "CLOSED"  # CLOSED, OPEN, HALF_OPEN
        
    def before_execution(self) -> bool:
        """Check if execution is allowed"""
        if self.state == "OPEN":
            if self._should_try_reset():
                self.state = "HALF_OPEN"
                logger.info(f"Circuit breaker {self.name} moving to HALF_OPEN")
                return True
            logger.warning(f"Circuit breaker {self.name} is OPEN")
            return False
        return True
        
    def after_success(self):
        """Handle successful execution"""
        if self.state == "HALF_OPEN":
            self.reset()
        self.failure_count = 0
        
    def after_failure(self):
        """Handle failed execution"""
        self.failure_count += 1
        self.last_failure_time = datetime.now()
        
        if self.failure_count >= self.failure_threshold:
            self.state = "OPEN"
            logger.error(f"Circuit breaker {self.name} triggered to OPEN state")
            
    def reset(self):
        """Reset circuit breaker"""
        self.state = "CLOSED"
        self.failure_count = 0
        self.last_failure_time = None
        logger.info(f"Circuit breaker {self.name} reset to CLOSED")
        
    def _should_try_reset(self) -> bool:
        """Check if we should attempt reset"""
        if not self.last_failure_time:
            return True
        time_since_failure = datetime.now() - self.last_failure_time
        return time_since_failure.total_seconds() >= self.reset_timeout
        
    def get_status(self) -> Dict[str, Any]:
        return {
            'name': self.name,
            'state': self.state,
            'failure_count': self.failure_count,
            'last_failure': self.last_failure_time
        }

def task_monitor(metrics: FailureMetrics, circuit_breaker: Optional[CircuitBreaker] = None):
    """Decorator to monitor task execution"""
    def decorator(func):
        @wraps(func)
        def wrapper(*args, **kwargs):
            task_name = func.__name__
            start_time = time.time()
            
            # Check circuit breaker
            if circuit_breaker and not circuit_breaker.before_execution():
                raise Exception(f"Circuit breaker {circuit_breaker.name} is OPEN")
                
            try:
                result = func(*args, **kwargs)
                duration = time.time() - start_time
                
                # Record success
                metrics.record_success(task_name, duration)
                if circuit_breaker:
                    circuit_breaker.after_success()
                    
                logger.info(f"Task {task_name} completed successfully in {duration:.2f}s")
                return result
                
            except Exception as e:
                duration = time.time() - start_time
                error_msg = str(e)
                
                # Record failure
                metrics.record_failure(task_name, error_msg, duration)
                if circuit_breaker:
                    circuit_breaker.after_failure()
                    
                logger.error(f"Task {task_name} failed after {duration:.2f}s: {error_msg}")
                raise
                
        return wrapper
    return decorator

def execute_with_retry(func, max_retries: int = 3, 
                       base_delay: float = 1.0,
                       backoff_factor: float = 2.0):
    """Execute function with retry logic"""
    last_exception = None
    
    for attempt in range(max_retries + 1):
        try:
            return func()
        except Exception as e:
            last_exception = e
            
            if attempt == max_retries:
                break
                
            delay = base_delay * (backoff_factor ** attempt)
            logger.warning(f"Attempt {attempt + 1} failed, retrying in {delay:.1f}s: {e}")
            time.sleep(delay)
            
    raise last_exception
```

def generate_hourly_chart(distribution):
    """Generate ASCII chart of hourly distribution"""
    max_count = max(distribution.values()) if distribution else 1
    chart = ""
    
    for hour in range(24):
        count = distribution.get(hour, 0)
        bar_length = int((count / max_count) * 50) if max_count > 0 else 0
        bar = "█" * bar_length
        chart += f"{hour:02d}:00 {bar} {count}\n"
        
    return chart
