import os
import json
import datetime
from pathlib import Path

def audit_system_health():
    """Comprehensive system health audit focusing on restoration needs"""
    
    audit_report = {
        "timestamp": datetime.datetime.now().isoformat(),
        "auditor": "spiritual_backbone",
        "virtue_applied": "LOVELY",
        "focus": "Seeking restoration in system health",
        "findings": [],
        "recommendations": [],
        "artifacts_created": []
    }
    
    # Check critical directories
    critical_dirs = [
        "agents/",
        "docs/",
        "logs/",
        "memory/",
        "projects/"
    ]
    
    print("🔍 Starting System Health Audit...")
    
    # 1. Directory Structure Health
    audit_report["findings"].append("=== DIRECTORY STRUCTURE AUDIT ===")
    for dir_path in critical_dirs:
        if os.path.exists(dir_path):
            try:
                files = os.listdir(dir_path)
                audit_report["findings"].append(f"✓ {dir_path}: {len(files)} items")
                
                # Check for empty directories that need attention
                if len(files) == 0:
                    audit_report["recommendations"].append(
                        f"RESTORE: Directory {dir_path} is empty - consider populating with templates"
                    )
            except Exception as e:
                audit_report["findings"].append(f"✗ {dir_path}: Error accessing - {str(e)}")
                audit_report["recommendations"].append(
                    f"FIX: Permission/access issue in {dir_path}"
                )
        else:
            audit_report["findings"].append(f"✗ {dir_path}: Missing")
            audit_report["recommendations"].append(
                f"RESTORE: Create missing directory {dir_path}"
            )
    
    # 2. Check for recent activity logs
    audit_report["findings"].append("\n=== ACTIVITY LOGS AUDIT ===")
    logs_dir = "logs/"
    if os.path.exists(logs_dir):
        log_files = [f for f in os.listdir(logs_dir) if f.endswith('.log') or f.endswith('.json')]
        if log_files:
            recent_logs = sorted(log_files, reverse=True)[:3]
            audit_report["findings"].append(f"Recent log files: {', '.join(recent_logs)}")
            
            # Check last modification time
            for log_file in recent_logs[:2]:
                log_path = os.path.join(logs_dir, log_file)
                mod_time = datetime.datetime.fromtimestamp(os.path.getmtime(log_path))
                hours_ago = (datetime.datetime.now() - mod_time).total_seconds() / 3600
                audit_report["findings"].append(f"  {log_file}: Modified {hours_ago:.1f} hours ago")
                
                if hours_ago > 24:
                    audit_report["recommendations"].append(
                        f"RESTORE: Log file {log_file} hasn't been updated in {hours_ago:.1f} hours"
                    )
        else:
            audit_report["findings"].append("No log files found")
            audit_report["recommendations"].append(
                "RESTORE: Create logging system to track swarm activity"
            )
    else:
        audit_report["recommendations"].append(
            "RESTORE: Create logs/ directory for system monitoring"
        )
    
    # 3. Check agent health
    audit_report["findings"].append("\n=== AGENT HEALTH AUDIT ===")
    agents_dir = "agents/"
    if os.path.exists(agents_dir):
        agent_files = [f for f in os.listdir(agents_dir) if f.endswith('.py') or f.endswith('.json')]
        audit_report["findings"].append(f"Total agents: {len(agent_files)}")
        
        # Check for core agents
        core_agents = ["trinity-apm", "trinity-mel", "spiritual_backbone"]
        found_agents = []
        for agent in core_agents:
            agent_patterns = [f for f in agent_files if agent in f.lower()]
            if agent_patterns:
                found_agents.append(agent)
                audit_report["findings"].append(f"  ✓ {agent}: Found")
            else:
                audit_report["findings"].append(f"  ✗ {agent}: Missing")
                audit_report["recommendations"].append(
                    f"RESTORE: Core agent {agent} appears to be missing"
                )
        
        audit_report["findings"].append(f"Core agents found: {len(found_agents)}/{len(core_agents)}")
    else:
        audit_report["recommendations"].append(
            "RESTORE: Create agents/ directory for swarm organization"
        )
    
    # 4. Check documentation completeness
    audit_report["findings"].append("\n=== DOCUMENTATION AUDIT ===")
    docs_dir = "docs/"
    essential_docs = [
        "STARTUP_DOCTRINE.md",
        "CONSTITUTION.md",
        "AGENT_PROTOCOLS.md",
        "MEMORY_SYSTEM.md"
    ]
    
    if os.path.exists(docs_dir):
        existing_docs = os.listdir(docs_dir)
        for doc in essential_docs:
            if doc in existing_docs:
                doc_path = os.path.join(docs_dir, doc)
                size = os.path.getsize(doc_path)
                audit_report["findings"].append(f"  ✓ {doc}: {size} bytes")
                
                if size < 100:  # Very small file
                    audit_report["recommendations"].append(
                        f"RESTORE: Documentation {doc} is very small ({size} bytes) - may need expansion"
                    )
            else:
                audit_report["findings"].append(f"  ✗ {doc}: Missing")
                audit_report["recommendations"].append(
                    f"RESTORE: Essential documentation {doc} is missing"
                )
    else:
        audit_report["recommendations"].append(
            "RESTORE: Create docs/ directory for system documentation"
        )
    
    # 5. Memory system check
    audit_report["findings"].append("\n=== MEMORY SYSTEM AUDIT ===")
    memory_dir = "memory/"
    if os.path.exists(memory_dir):
        memory_files = os.listdir(memory_dir)
        audit_report["findings"].append(f"Memory files: {len(memory_files)}")
        
        # Check for recent memories
        if memory_files:
            recent_memories = sorted(memory_files, reverse=True)[:5]
            audit_report["findings"].append(f"Recent memory files: {', '.join(recent_memories[:3])}")
        else:
            audit_report["findings"].append("No memory files found")
            audit_report["recommendations"].append(
                "RESTORE: Memory system appears unused - implement memory persistence"
            )
    else:
        audit_report["recommendations"].append(
            "RESTORE: Create memory/ directory for swarm memory persistence"
        )
    
    # 6. Project tracking
    audit_report["findings"].append("\n=== PROJECT TRACKING AUDIT ===")
    projects_dir = "projects/"
    if os.path.exists(projects_dir):
        projects = os.listdir(projects_dir)
        active_projects = []
        
        for project in projects:
            project_path = os.path.join(projects_dir, project)
            if os.path.isdir(project_path):
                # Check for recent activity
                try:
                    files_in_project = os.listdir(project_path)
                    if files_in_project:
                        # Get most recent file modification
                        recent_files = sorted(
                            [os.path.join(project_path, f) for f in files_in_project],
                            key=os.path.getmtime,
                            reverse=True
                        )[:3]
                        
                        if recent_files:
                            latest_mod = datetime.datetime.fromtimestamp(
                                os.path.getmtime(recent_files[0])
                            )
                            days_since = (datetime.datetime.now() - latest_mod).days
                            
                            if days_since < 7:
                                active_projects.append(project)
                                status = "ACTIVE"
                            elif days_since < 30:
                                status = "STALE"
                            else:
                                status = "DORMANT"
                            
                            audit_report["findings"].append(
                                f"  {project}: {status} ({len(files_in_project)} files, {days_since}d ago)"
                            )
                            
                            if status in ["STALE", "DORMANT"]:
                                audit_report["recommendations"].append(
                                    f"RESTORE: Project {project} is {status.lower()} - consider review or archival"
                                )
                except Exception as e:
                    audit_report["findings"].append(f"  {project}: Error - {str(e)}")
        
        audit_report["findings"].append(f"Active projects: {len(active_projects)}/{len(projects)}")
        
        if len(active_projects) == 0 and len(projects) > 0:
            audit_report["recommendations"].append(
                "RESTORE: No active projects found - swarm may be idle"
            )
    else:
        audit_report["recommendations"].append(
            "RESTORE: Create projects/ directory for work organization"
        )
    
    # 7. System performance indicators
    audit_report["findings"].append("\n=== PERFORMANCE INDICATORS ===")
    
    # Check disk space (simulated - would need proper system calls)
    try:
        import shutil
        total, used, free = shutil.disk_usage(".")
        free_gb = free / (1024**3)
        audit_report["findings"].append(f"Disk space free: {free_gb:.1f} GB")
        
        if free_gb < 1:
            audit_report["recommendations"].append(
                "RESTORE: Low disk space (<1GB free) - consider cleanup"
            )
    except:
        audit_report["findings"].append("Disk space: Unable to check")
    
    # 8. Generate health score
    total_checks = len([f for f in audit_report["findings"] if "✓" in f or "✗" in f])
    passed_checks = len([f for f in audit_report["findings"] if "✓" in f])
    
    if total_checks > 0:
        health_score = (passed_checks / total_checks) * 100
    else:
        health_score = 0
    
    audit_report["health_score"] = f"{health_score:.1f}%"
    audit_report["summary"] = f"System Health: {health_score:.1f}% - {len(audit_report['recommendations'])} restoration actions needed"
    
    # Add timestamp for tracking
    audit_report["audit_completed"] = datetime.datetime.now().isoformat()
    
    return audit_report

# Execute the audit
