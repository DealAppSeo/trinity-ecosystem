from supabase import create_client, Client
import pandas as pd

# Process the result into a pandas DataFrame for easier analysis

    # Simplified example: Check if the column is used in the last 100 queries
    queries = supabase.from_("query_logs").select("query").limit(100).execute()
    if column not in str(queries.data):
        unused_columns.append(column)

# Step 4: Document findings
with open("trinity_tasks_audit.md", "w") as file:
