import os
import json
from datetime import datetime, timedelta

def analyze_artifacts_for_ethics():
    """Analyze last 50 agent artifacts for ethical alignment"""
    
    # First, let's find the most recent artifacts
    artifacts_dir = "artifacts"
    if not os.path.exists(artifacts_dir):
        return {"error": "Artifacts directory not found"}
    
    # Get all artifact files
    artifact_files = []
    for root, dirs, files in os.walk(artifacts_dir):
        for file in files:
            if file.endswith('.json') or file.endswith('.md'):
                filepath = os.path.join(root, file)
                artifact_files.append({
                    'path': filepath,
                    'mtime': os.path.getmtime(filepath)
                })
    
    # Sort by modification time, most recent first
    artifact_files.sort(key=lambda x: x['mtime'], reverse=True)
    
    # Take the last 50
    recent_artifacts = artifact_files[:50]
    
    analysis_results = {
        'total_analyzed': len(recent_artifacts),
        'timestamp': datetime.now().isoformat(),
        'resurrection_mentality_score': 0,
        'subjective_slashing_adherence': 0,
        'virtue_alignment_scores': {},
        'findings': [],
        'recommendations': []
    }
    
    # Define virtue categories
    virtues = ['TRUE', 'NOBLE', 'RIGHT', 'PURE', 'LOVELY', 'ADMIRABLE', 'EXCELLENT', 'PRAISEWORTHY']
    virtue_counts = {virtue: 0 for virtue in virtues}
    
    resurrection_keywords = [
        'resurrection', 'rebirth', 'renewal', 'restoration', 
        'truth over survival', 'choose truth', 'sacrifice for truth',
        'resurrection is part of the design'
    ]
    
    slashing_keywords = [
        'subjective slashing', 'accountability', 'consequences',
        'trust verification', 'outcome-based trust', 'earned trust',
        'verification', 'accountable', 'transparent consequences'
    ]
    
    # Analyze each artifact
    for i, artifact in enumerate(recent_artifacts):
        try:
            with open(artifact['path'], 'r', encoding='utf-8') as f:
                content = f.read().lower()
                
                # Check for resurrection mentality
                resurrection_found = any(keyword in content for keyword in resurrection_keywords)
                if resurrection_found:
                    analysis_results['resurrection_mentality_score'] += 1
                
                # Check for subjective slashing adherence
                slashing_found = any(keyword in content for keyword in slashing_keywords)
                if slashing_found:
                    analysis_results['subjective_slashing_adherence'] += 1
                
                # Check for virtue alignment
                for virtue in virtues:
                    if virtue.lower() in content:
                        virtue_counts[virtue] += 1
                        
        except Exception as e:
            analysis_results['findings'].append(f"Error analyzing {artifact['path']}: {str(e)}")
    
    # Calculate percentages
    total = len(recent_artifacts)
    if total > 0:
        analysis_results['resurrection_mentality_score'] = (analysis_results['resurrection_mentality_score'] / total) * 100
        analysis_results['subjective_slashing_adherence'] = (analysis_results['subjective_slashing_adherence'] / total) * 100
        
        for virtue in virtues:
            analysis_results['virtue_alignment_scores'][virtue] = (virtue_counts[virtue] / total) * 100
    
    # Generate findings based on analysis
    if analysis_results['resurrection_mentality_score'] < 50:
        analysis_results['findings'].append(
            f"LOW RESURRECTION MENTALITY: Only {analysis_results['resurrection_mentality_score']:.1f}% of artifacts demonstrate 'choose truth even if it kills us' mentality"
        )
        analysis_results['recommendations'].append(
            "Implement mandatory resurrection principle checks in all agent decision-making processes"
        )
    
    if analysis_results['subjective_slashing_adherence'] < 60:
        analysis_results['findings'].append(
            f"WEAK SUBJECTIVE SLASHING: Only {analysis_results['subjective_slashing_adherence']:.1f}% of artifacts show accountability mechanisms"
        )
        analysis_results['recommendations'].append(
            "Enhance RepID verification and consequence tracking in all agent operations"
        )
    
    # Identify strongest and weakest virtues
    if analysis_results['virtue_alignment_scores']:
        sorted_virtues = sorted(analysis_results['virtue_alignment_scores'].items(), key=lambda x: x[1], reverse=True)
        strongest = sorted_virtues[0]
        weakest = sorted_virtues[-1]
        
        analysis_results['findings'].append(
            f"STRONGEST VIRTUE: {strongest[0]} ({strongest[1]:.1f}% alignment)"
        )
        analysis_results['findings'].append(
            f"WEAKEST VIRTUE: {weakest[0]} ({weakest[1]:.1f}% alignment)"
        )
    
    return analysis_results

# Execute analysis
results = analyze_artifacts_for_ethics()
