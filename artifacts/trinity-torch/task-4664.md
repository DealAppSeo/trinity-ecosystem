import os
import json
import datetime

# First, let me examine the current directory structure to understand the ecosystem
def audit_system_structure():
    """Audit the current system structure and swarm organization"""
    
    print("🔍 Conducting System Structure Audit...")
    
    # Define key directories to check
    key_dirs = [
        "docs/",
        "agents/",
        "projects/",
        "logs/",
        "artifacts/",
        "specs/"
    ]
    
    # Define key files to check
    key_files = [
        "docs/CONSTITUTION.md",
        "docs/STARTUP_DOCTRINE.md",
        "docs/AGENT_PROTOCOLS.md",
        "agents/registry.json",
        "projects/active_projects.json"
    ]
    
    findings = {
        "timestamp": datetime.datetime.now().isoformat(),
        "system_audit": {
            "directory_health": {},
            "file_health": {},
            "swarm_status": {},
            "issues_found": [],
            "recommendations": []
        }
    }
    
    # Check directory health
    for dir_path in key_dirs:
        if os.path.exists(dir_path):
            try:
                contents = os.listdir(dir_path)
                findings["system_audit"]["directory_health"][dir_path] = {
                    "exists": True,
                    "item_count": len(contents),
                    "last_modified": datetime.datetime.fromtimestamp(
                        os.path.getmtime(dir_path)
                    ).isoformat() if contents else "N/A"
                }
            except Exception as e:
                findings["system_audit"]["directory_health"][dir_path] = {
                    "exists": True,
                    "error": str(e)
                }
        else:
            findings["system_audit"]["directory_health"][dir_path] = {
                "exists": False,
                "warning": "Directory missing"
            }
            findings["system_audit"]["issues_found"].append(
                f"Missing directory: {dir_path}"
            )
    
    # Check file health
    for file_path in key_files:
        if os.path.exists(file_path):
            try:
                file_stats = os.stat(file_path)
                findings["system_audit"]["file_health"][file_path] = {
                    "exists": True,
                    "size_bytes": file_stats.st_size,
                    "last_modified": datetime.datetime.fromtimestamp(
                        file_stats.st_mtime
                    ).isoformat(),
                    "last_accessed": datetime.datetime.fromtimestamp(
                        file_stats.st_atime
                    ).isoformat()
                }
            except Exception as e:
                findings["system_audit"]["file_health"][file_path] = {
                    "exists": True,
                    "error": str(e)
                }
        else:
            findings["system_audit"]["file_health"][file_path] = {
                "exists": False,
                "warning": "File missing"
            }
            findings["system_audit"]["issues_found"].append(
                f"Missing critical file: {file_path}"
            )
    
    # Check for agent registry
    if os.path.exists("agents/registry.json"):
        try:
            with open("agents/registry.json", "r") as f:
                registry = json.load(f)
                findings["system_audit"]["swarm_status"]["agent_count"] = len(registry.get("agents", []))
                findings["system_audit"]["swarm_status"]["active_agents"] = [
                    agent for agent in registry.get("agents", [])
                    if agent.get("status") == "active"
                ]
                findings["system_audit"]["swarm_status"]["registry_timestamp"] = registry.get("last_updated", "unknown")
        except Exception as e:
            findings["system_audit"]["swarm_status"]["registry_error"] = str(e)
            findings["system_audit"]["issues_found"].append(
                f"Failed to parse agent registry: {e}"
            )
    else:
        findings["system_audit"]["swarm_status"]["warning"] = "No agent registry found"
        findings["system_audit"]["issues_found"].append("Missing agent registry")
    
    # Check for recent activity in logs
    log_files = []
    if os.path.exists("logs/"):
        try:
            log_files = [f for f in os.listdir("logs/") if f.endswith('.log')]
            recent_logs = []
            for log_file in log_files[:5]:  # Check most recent 5
                log_path = os.path.join("logs/", log_file)
                mod_time = datetime.datetime.fromtimestamp(os.path.getmtime(log_path))
                recent_logs.append({
                    "file": log_file,
                    "last_modified": mod_time.isoformat(),
                    "age_days": (datetime.datetime.now() - mod_time).days
                })
            findings["system_audit"]["swarm_status"]["recent_logs"] = recent_logs
            findings["system_audit"]["swarm_status"]["total_log_files"] = len(log_files)
        except Exception as e:
            findings["system_audit"]["swarm_status"]["log_error"] = str(e)
    
    # Generate recommendations based on findings
    if not findings["system_audit"]["issues_found"]:
        findings["system_audit"]["recommendations"].append(
            "System structure appears healthy. Continue regular maintenance."
        )
    else:
        findings["system_audit"]["recommendations"].append(
            "Address missing directories and files identified in issues_found."
        )
    
    # Check for recent artifact creation
    if os.path.exists("artifacts/"):
        try:
            artifacts = [f for f in os.listdir("artifacts/") if not f.startswith('.')]
            if artifacts:
                latest_artifact = max(
                    artifacts,
                    key=lambda x: os.path.getmtime(os.path.join("artifacts/", x))
                )
                artifact_time = datetime.datetime.fromtimestamp(
                    os.path.getmtime(os.path.join("artifacts/", latest_artifact))
                )
                findings["system_audit"]["swarm_status"]["latest_artifact"] = {
                    "name": latest_artifact,
                    "created": artifact_time.isoformat(),
                    "age_hours": (datetime.datetime.now() - artifact_time).total_seconds() / 3600
                }
                findings["system_audit"]["swarm_status"]["total_artifacts"] = len(artifacts)
        except Exception as e:
            findings["system_audit"]["swarm_status"]["artifact_error"] = str(e)
    
    return findings

# Execute the audit
