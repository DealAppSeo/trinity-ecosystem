def calculate_health_score(architecture, performance, compliance, resources):
    """Calculate overall health score from audit components"""
    
    # Weighted scoring
    weights = {
        "architecture": 0.25,
        "performance": 0.30,
        "compliance": 0.25,
        "resources": 0.20
    }
    
    # Architecture score (based on active modules)
    arch_score = sum(1 for module in architecture["core_modules"] if module["status"] == "ACTIVE") / len(architecture["core_modules"])
    
    # Performance score (based on healthy endpoints)
    perf_score = sum(1 for endpoint in performance["response_times"] if endpoint["status"] == "HEALTHY") / len(performance["response_times"])
    
    # Compliance score (based on compliant virtues)
    comp_score = sum(1 for virtue in compliance["eight_virtues"].values() if virtue["status"] == "COMPLIANT") / len(compliance["eight_virtues"])
    
    # Resource score (based on utilization thresholds)
    resource_metrics = resources["resource_allocation"]["compute_resources"]
    util_scores = [
        1 - (resource_metrics["cpu_utilization"] / 100) if resource_metrics["cpu_utilization"] < 80 else 0.2,
        1 - (resource_metrics["memory_utilization"] / 100) if resource_metrics["memory_utilization"] < 85 else 0.2,
        1 - (resource_metrics["disk_utilization"] / 100) if resource_metrics["disk_utilization"] < 90 else 0.2,
        min(resource_metrics["network_bandwidth"] / 500, 1)  # Assuming 500mbps is optimal
    ]
    res_score = sum(util_scores) / len(util_scores)
    
    # Calculate weighted total
    total_score = (
        arch_score * weights["architecture"] +
        perf_score * weights["performance"] +
        comp_score * weights["compliance"] +
        res_score * weights["resources"]
    )
    
    return round(total_score * 100, 2)
