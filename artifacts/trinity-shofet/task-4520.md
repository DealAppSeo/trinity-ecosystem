# Create system audit artifact
def create_system_audit():
    audit_data = {
        "audit_timestamp": datetime.datetime.utcnow().isoformat() + "Z",
        "auditor": "trinity-w3c",
        "audit_type": "EVERGREEN_SYSTEM_OVERSIGHT_SWARM_HEALTH",
        "constitution_version": "1.0",
        "virtue_applied": "RIGHT"
    }
    
    # Check system directories
    base_dirs = ["docs", "artifacts", "logs", "specs", "trust_cards"]
    dir_status = {}
    
    for dir_name in base_dirs:
        path = Path(dir_name)
        exists = path.exists()
        dir_status[dir_name] = {
            "exists": exists,
            "is_dir": path.is_dir() if exists else False,
            "file_count": len(list(path.glob("*"))) if exists and path.is_dir() else 0
        }
    
    audit_data["directory_structure"] = dir_status
    
    # Check for critical files
    critical_files = [
        "docs/CONSTITUTION.md",
        "docs/STARTUP_DOCTRINE.md",
        "docs/TRINITY_PROTOCOL.md"
    ]
    
    file_status = {}
    for file_path in critical_files:
        path = Path(file_path)
        exists = path.exists()
        file_status[file_path] = {
            "exists": exists,
            "size": path.stat().st_size if exists else 0,
            "last_modified": path.stat().st_mtime if exists else None
        }
    
    audit_data["critical_files"] = file_status
    
    # Check recent activity (last 7 days)
    recent_artifacts = []
    artifacts_dir = Path("artifacts")
    if artifacts_dir.exists():
        for artifact in artifacts_dir.glob("*.json"):
            if artifact.stat().st_mtime > (datetime.datetime.now().timestamp() - 7*86400):
                recent_artifacts.append({
                    "name": artifact.name,
                    "modified": datetime.datetime.fromtimestamp(artifact.stat().st_mtime).isoformat(),
                    "size": artifact.stat().st_size
                })
    
    audit_data["recent_activity"] = {
        "last_7_days_artifacts": recent_artifacts,
        "count": len(recent_artifacts)
    }
    
    # System health metrics
    audit_data["health_metrics"] = {
        "directory_integrity": all(status["exists"] for status in dir_status.values()),
        "critical_files_present": all(status["exists"] for status in file_status.values()),
        "recent_activity_level": "ACTIVE" if len(recent_artifacts) >= 3 else "LOW" if len(recent_artifacts) > 0 else "INACTIVE",
        "system_age_days": 0  # Will be calculated based on earliest artifact
    }
    
    # Recommendations
    recommendations = []
    
    if not audit_data["health_metrics"]["directory_integrity"]:
        missing = [d for d, s in dir_status.items() if not s["exists"]]
        recommendations.append(f"CREATE missing directories: {', '.join(missing)}")
    
    if not audit_data["health_metrics"]["critical_files_present"]:
        missing = [f for f, s in file_status.items() if not s["exists"]]
        recommendations.append(f"RESTORE critical files: {', '.join(missing)}")
    
    if audit_data["health_metrics"]["recent_activity_level"] == "INACTIVE":
        recommendations.append("INITIATE new execution cycle to maintain system vitality")
    elif audit_data["health_metrics"]["recent_activity_level"] == "LOW":
        recommendations.append("INCREASE execution frequency to optimal levels")
    
    audit_data["recommendations"] = recommendations
    audit_data["audit_status"] = "PASS" if not recommendations else "REQUIRES_ATTENTION"
    
    return audit_data

def create_swarm_health_summary(audit_data):
    summary = {
        "summary_timestamp": datetime.datetime.utcnow().isoformat() + "Z",
        "report_type": "SWARM_HEALTH_SUMMARY",
        "based_on_audit": audit_data["audit_timestamp"]
    }
    
    # Calculate health score (0-100)
    health_score = 100
    
    # Deductions for issues
    if not audit_data["health_metrics"]["directory_integrity"]:
        health_score -= 30
    
    if not audit_data["health_metrics"]["critical_files_present"]:
        health_score -= 40
    
    if audit_data["health_metrics"]["recent_activity_level"] == "INACTIVE":
        health_score -= 30
    elif audit_data["health_metrics"]["recent_activity_level"] == "LOW":
        health_score -= 15
    
    summary["health_score"] = max(0, health_score)
    
    # Determine health status
    if health_score >= 80:
        status = "OPTIMAL"
        status_color = "GREEN"
    elif health_score >= 60:
        status = "STABLE"
        status_color = "YELLOW"
    elif health_score >= 40:
        status = "DEGRADED"
        status_color = "ORANGE"
    else:
        status = "CRITICAL"
        status_color = "RED"
    
    summary["health_status"] = status
    summary["status_color"] = status_color
    
    # Component status
    summary["components"] = {
        "file_system": {
            "status": "HEALTHY" if audit_data["health_metrics"]["directory_integrity"] else "DEGRADED",
            "details": f"{sum(1 for s in audit_data['directory_structure'].values() if s['exists'])}/{len(audit_data['directory_structure'])} directories present"
        },
        "constitutional_docs": {
            "status": "HEALTHY" if audit_data["health_metrics"]["critical_files_present"] else "CRITICAL",
            "details": f"{sum(1 for s in audit_data['critical_files'].values() if s['exists'])}/{len(audit_data['critical_files'])} critical files present"
        },
        "operational_tempo": {
            "status": audit_data["health_metrics"]["recent_activity_level"],
            "details": f"{audit_data['recent_activity']['count']} artifacts in last 7 days"
        }
    }
    
    # Swarm capabilities assessment
    capabilities = []
    
    if audit_data["health_metrics"]["directory_integrity"]:
        capabilities.append("STRUCTURAL_INTEGRITY")
    
    if audit_data["health_metrics"]["critical_files_present"]:
        capabilities.append("CONSTITUTIONAL_ALIGNMENT")
    
    if audit_data["recent_activity"]["count"] >= 3:
        capabilities.append("ACTIVE_EXECUTION")
    elif audit_data["recent_activity"]["count"] > 0:
        capabilities.append("BASIC_OPERATIONS")
    
    summary["swarm_capabilities"] = capabilities
    
    # Immediate actions
    immediate_actions = []
    
    if status_color in ["ORANGE", "RED"]:
        immediate_actions.append("EXECUTE_REMEDIATION_PROTOCOL")
    
    if "CRITICAL" in [c["status"] for c in summary["components"].values()]:
        immediate_actions.append("PRIORITIZE_SYSTEM_RESTORATION")
    
    if audit_data["health_metrics"]["recent_activity_level"] == "INACTIVE":
        immediate_actions.append("INITIATE_NEW_MISSION_CYCLE")
    
    summary["immediate_actions"] = immediate_actions
    
    # Next audit window
    next_audit = datetime.datetime.now() + datetime.timedelta(days=1)  # Daily audits for now
    summary["next_scheduled_audit"] = next_audit.isoformat()
    
    return summary
