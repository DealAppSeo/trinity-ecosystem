import os
import json
import datetime
from pathlib import Path

def audit_system_health():
    """Comprehensive system oversight and swarm health audit"""
    
    audit_report = {
        "timestamp": datetime.datetime.now().isoformat(),
        "auditor": "governance.Virtue:RIGHT",
        "audit_type": "EVERGREEN_System_Oversight_Swarm_Health",
        "status": "IN_PROGRESS"
    }
    
    # Check system directories and structure
    system_dirs = {
        "docs": "Documentation directory",
        "agents": "Agent implementations",
        "logs": "System logs",
        "artifacts": "Generated artifacts",
        "context": "Context files"
    }
    
    dir_health = {}
    for dir_name, description in system_dirs.items():
        dir_path = Path(dir_name)
        exists = dir_path.exists()
        is_dir = dir_path.is_dir() if exists else False
        file_count = len(list(dir_path.glob("*"))) if exists and is_dir else 0
        
        dir_health[dir_name] = {
            "exists": exists,
            "is_directory": is_dir,
            "file_count": file_count,
            "status": "HEALTHY" if exists and is_dir else "ISSUE"
        }
    
    audit_report["directory_structure"] = dir_health
    
    # Check for critical files
    critical_files = [
        "docs/CONSTITUTION.md",
        "docs/STARTUP_DOCTRINE.md",
        "context/trinity_context.md"
    ]
    
    file_health = {}
    for file_path in critical_files:
        path = Path(file_path)
        exists = path.exists()
        is_file = path.is_file() if exists else False
        size = path.stat().st_size if exists and is_file else 0
        
        file_health[file_path] = {
            "exists": exists,
            "is_file": is_file,
            "size_bytes": size,
            "status": "HEALTHY" if exists and is_file and size > 0 else "ISSUE"
        }
    
    audit_report["critical_files"] = file_health
    
    # Check recent activity (last 7 days)
    recent_artifacts = []
    artifacts_dir = Path("artifacts")
    if artifacts_dir.exists():
        for artifact in artifacts_dir.glob("**/*"):
            if artifact.is_file():
                mod_time = datetime.datetime.fromtimestamp(artifact.stat().st_mtime)
                age_days = (datetime.datetime.now() - mod_time).days
                if age_days <= 7:
                    recent_artifacts.append({
                        "path": str(artifact),
                        "modified": mod_time.isoformat(),
                        "age_days": age_days,
                        "size_bytes": artifact.stat().st_size
                    })
    
    audit_report["recent_activity"] = {
        "artifacts_last_7_days": len(recent_artifacts),
        "sample_artifacts": recent_artifacts[:5]  # First 5 as sample
    }
    
    # Check system performance metrics
    try:
        # Check for system logs
        log_files = list(Path("logs").glob("*.log")) if Path("logs").exists() else []
        
        # Check for agent activity
        agent_files = list(Path("agents").glob("*.py")) if Path("agents").exists() else []
        
        audit_report["performance_metrics"] = {
            "log_files_count": len(log_files),
            "active_agents": len(agent_files),
            "total_artifacts": len(list(artifacts_dir.glob("**/*"))) if artifacts_dir.exists() else 0,
            "system_uptime_estimate": "ACTIVE" if len(recent_artifacts) > 0 else "INACTIVE"
        }
    except Exception as e:
        audit_report["performance_metrics"] = {
            "error": str(e),
            "status": "CHECK_FAILED"
        }
    
    # Generate health score
    issues = []
    
    # Directory issues
    for dir_name, health in dir_health.items():
        if not health["exists"] or not health["is_directory"]:
            issues.append(f"Directory issue: {dir_name}")
    
    # File issues
    for file_path, health in file_health.items():
        if not health["exists"] or not health["is_file"] or health["size_bytes"] == 0:
            issues.append(f"File issue: {file_path}")
    
    # Activity issues
    if audit_report["recent_activity"]["artifacts_last_7_days"] == 0:
        issues.append("No recent activity detected")
    
    health_score = 100 - (len(issues) * 10)
    health_score = max(0, min(100, health_score))
    
    audit_report["health_assessment"] = {
        "total_issues": len(issues),
        "issues": issues,
        "health_score": health_score,
        "status": "HEALTHY" if health_score >= 80 else "WARNING" if health_score >= 60 else "CRITICAL"
    }
    
    # Recommendations
    recommendations = []
    if health_score < 80:
        recommendations.append("Immediate attention required for critical system components")
    if audit_report["recent_activity"]["artifacts_last_7_days"] < 3:
        recommendations.append("Increase system activity and monitoring")
    for issue in issues:
        if "CONSTITUTION.md" in issue:
            recommendations.append("Restore CONSTITUTION.md immediately")
        if "STARTUP_DOCTRINE.md" in issue:
            recommendations.append("Restore STARTUP_DOCTRINE.md for operational guidance")
    
    audit_report["recommendations"] = recommendations
    
    audit_report["status"] = "COMPLETED"
    
    return audit_report

# Execute audit
audit_result = audit_system_health()

# Save artifact
artifact_content = json.dumps(audit_result, indent=2)
artifact_filename = f"artifacts/system_health_audit_{datetime.datetime.now().strftime('%Y%m%d_%H%M%S')}.json"

# Ensure artifacts directory exists

    f.write(artifact_content)
