import os
import ast
import subprocess
from datetime import datetime

def analyze_codebase():
    """Analyze code quality metrics for the current project"""
    
    report = {
        "timestamp": datetime.now().isoformat(),
        "project": "Trinity-Shofet System",
        "analysis_date": datetime.now().strftime("%Y-%m-%d"),
        "metrics": {},
        "issues": [],
        "recommendations": []
    }
    
    # Check for key files and structure
    key_files = [
        "main.py",
        "constitution.py", 
        "docs/STARTUP_DOCTRINE.md",
        "requirements.txt",
        "README.md"
    ]
    
    existing_files = []
    for file in key_files:
        if os.path.exists(file):
            existing_files.append(file)
    
    report["file_structure"] = {
        "expected_files": len(key_files),
        "existing_files": len(existing_files),
        "missing_files": [f for f in key_files if f not in existing_files],
        "coverage_percentage": (len(existing_files) / len(key_files)) * 100
    }
    
    # Analyze Python files for basic quality metrics
    python_files = []
    for root, dirs, files in os.walk("."):
        for file in files:
            if file.endswith(".py"):
                python_files.append(os.path.join(root, file))
    
    report["python_analysis"] = {
        "total_files": len(python_files),
        "files_analyzed": [],
        "lines_of_code": 0,
        "functions": 0,
        "classes": 0,
        "imports": 0
    }
    
    # Basic analysis of Python files
    for py_file in python_files[:10]:  # Limit to first 10 for speed
        try:
            with open(py_file, 'r', encoding='utf-8') as f:
                content = f.read()
                lines = content.split('\n')
                
                report["python_analysis"]["lines_of_code"] += len([l for l in lines if l.strip()])
                report["python_analysis"]["files_analyzed"].append(py_file)
                
                # Parse AST for more detailed analysis
                try:
                    tree = ast.parse(content)
                    
                    # Count functions
                    functions = [node for node in ast.walk(tree) if isinstance(node, ast.FunctionDef)]
                    report["python_analysis"]["functions"] += len(functions)
                    
                    # Count classes
                    classes = [node for node in ast.walk(tree) if isinstance(node, ast.ClassDef)]
                    report["python_analysis"]["classes"] += len(classes)
                    
                    # Count imports
                    imports = [node for node in ast.walk(tree) if isinstance(node, (ast.Import, ast.ImportFrom))]
                    report["python_analysis"]["imports"] += len(imports)
                    
                except SyntaxError as e:
                    report["issues"].append(f"Syntax error in {py_file}: {str(e)}")
                    
        except Exception as e:
            report["issues"].append(f"Error reading {py_file}: {str(e)}")
    
    # Check for requirements.txt
    if os.path.exists("requirements.txt"):
        try:
            with open("requirements.txt", 'r') as f:
                requirements = [line.strip() for line in f if line.strip() and not line.startswith('#')]
                report["dependencies"] = {
                    "total": len(requirements),
                    "list": requirements[:10]  # First 10 only
                }
        except:
            report["dependencies"] = {"total": 0, "list": []}
    else:
        report["issues"].append("Missing requirements.txt file")
    
    # Check for README
    if not os.path.exists("README.md"):
        report["issues"].append("Missing README.md file")
        report["recommendations"].append("Create comprehensive README with setup instructions")
    
    # Check for tests
    test_files = [f for f in python_files if 'test' in f.lower()]
    report["testing"] = {
        "test_files_found": len(test_files),
        "test_coverage": "Unknown"  # Would need pytest-cov for actual coverage
    }
    
    if len(test_files) == 0:
        report["issues"].append("No test files found")
        report["recommendations"].append("Add unit tests for core functionality")
    
    # Calculate quality score (simplified)
    quality_score = 0
    
    # File structure contributes 30%
    quality_score += report["file_structure"]["coverage_percentage"] * 0.3
    
    # Code metrics contribute 40%
    if report["python_analysis"]["total_files"] > 0:
        avg_lines_per_file = report["python_analysis"]["lines_of_code"] / max(1, len(report["python_analysis"]["files_analyzed"]))
        if avg_lines_per_file < 200:  # Good if files are reasonably sized
            quality_score += 40
        elif avg_lines_per_file < 500:
            quality_score += 20
        else:
            quality_score += 10
    
    # Testing contributes 30%
    if report["testing"]["test_files_found"] > 0:
        quality_score += 30
    elif report["python_analysis"]["total_files"] > 0:
        quality_score += 10  # Partial credit for having code
    
    report["quality_score"] = min(100, quality_score)
    
    # Determine quality grade
    if report["quality_score"] >= 80:
        report["quality_grade"] = "A - Excellent"
    elif report["quality_score"] >= 70:
        report["quality_grade"] = "B - Good"
    elif report["quality_score"] >= 60:
        report["quality_grade"] = "C - Fair"
    elif report["quality_score"] >= 50:
        report["quality_grade"] = "D - Needs Improvement"
    else:
        report["quality_grade"] = "F - Critical Issues"
    
    return report

# Generate the report
