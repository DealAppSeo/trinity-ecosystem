// Function to audit recent RepID updates
function auditRepIDUpdates(userActivities, baselineScore = 0) {
  // Initialize finalScore with a baseline value
  let finalScore = baselineScore;

  // Calculate score adjustments based on user activities
  if (userActivities && userActivities.length > 0) {
    userActivities.forEach(activity => {
      if (activity.type === 'successful_run') {
        finalScore += 10; // Increment for verified successful runs
      } else if (activity.type === 'failed_run') {
        finalScore -= 5; // Decrement for failures
      }
    });
  } else {
    console.log('No activities found for RepID audit.');
  }

  // Ensure finalScore is non-negative
  finalScore = Math.max(0, finalScore);

  // Return audit result
  return {
    userId: userActivities[0]?.userId || 'unknown',
    finalScore: finalScore,
    auditDate: new Date().toISOString(),
    activityCount: userActivities.length
  };
}

// Example usage
const sampleActivities = [
  { userId: 'user123', type: 'successful_run' },
  { userId: 'user123', type: 'successful_run' },
  { userId: 'user123', type: 'failed_run' }
];

const auditResult = auditRepIDUpdates(sampleActivities, 50);
console.log('Audit Result:', auditResult);
