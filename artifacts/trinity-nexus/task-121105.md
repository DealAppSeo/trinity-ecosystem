import { test, expect } from '@playwright/test';

test.describe('Global Heartbeat', () => {
  test('Heartbeat endpoint responds within 500ms', async ({ page, request }) => {
    const response = await request.get('/api/health');
    expect(response.ok()).toBeTruthy();
    expect(response.status()).toBe(200);
    
    const data = await response.json();
    expect(data).toHaveProperty('status', 'healthy');
    expect(data).toHaveProperty('timestamp');
    expect(data.uptime).toBeGreaterThan(0);
  });

  test('State management persists across restarts', async ({ page }) => {
    // Verify state restoration
    await page.goto('/dashboard');
    const initialState = await page.locator('[data-state]').textContent();
    
    // Simulate restart
    await page.reload();
    const restoredState = await page.locator('[data-state]').textContent();
    
    expect(restoredState).toEqual(initialState);
  });
});
