function calculateFinalScore(investmentValue) {
  if (investmentValue > 1000) {
    // high-risk investment with high potential reward
    return investmentValue * 2;
  } else if (investmentValue > 500) {
    // moderate-risk investment with moderate potential reward
    return investmentValue * 1.5;
  } else {
    // low-risk investment with low potential reward
    return investmentValue * 1;
  }
}

function captureInvestmentValue(value) {
  console.log(`Captured investment value: ${value}`);
}

function calculateInvestmentValue(values) {
  let investmentValue = 0;
  for (const value of values) {
    investmentValue += value;
  }
  return investmentValue;
}

const values = [500, 200, 300];
const investmentValue = calculateInvestmentValue(values);
captureInvestmentValue(investmentValue);
const finalScore = calculateFinalScore(investmentValue);
console.log(`Final Score: ${finalScore}`);

import { chromium } from 'playwright';

describe('Final Score Calculator', () => {
  it('should calculate the final score', async () => {
    const browser = await chromium.launch();
    const page = await browser.newPage();
    await page.goto('https://example.com/calculator');
    await page.type('[id="investment-value"]', '500');
    await page.press('[id="investment-value"]', 'Tab');
    await page.type('[id="calculate-investment-value"]', 'Calculate');
    await page.waitForSelector('[id="final-score"]');
    const finalScore = await page.$eval('#final-score', (el) => el.textContent);
    expect(finalScore).toBe('1000');
    await browser.close();
  });
});
