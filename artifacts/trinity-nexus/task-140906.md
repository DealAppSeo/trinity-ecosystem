/**
 * Merkle Hash Utility for Trinity-Nexus Artifacts
 * Implements SHA2-256 multihash for artifact integrity and identification.
 * Designed for integration with artifact write operations while maintaining
 * backward compatibility with existing artifact IDs.
 * 
 * @module merkle
 */

const crypto = require('crypto');

/**
 * Computes a SHA2-256 hash of the given content.
 * This function is used to generate a unique identifier for artifacts
 * based on their content. It ensures data integrity and supports
 * Merkle tree structures for hierarchical validation.
 * 
 * @param {string|Buffer} content - The content to hash (string or Buffer).
 * @returns {string} - The hexadecimal representation of the SHA2-256 hash.
 * @throws {Error} - If content is invalid or hash computation fails.
 */
function computeSHA256(content) {
  try {
    if (!content) {
      throw new Error('Content is empty or undefined');
    }
    const hash = crypto.createHash('sha256');
    hash.update(content);
    return hash.digest('hex');
  } catch (error) {
    throw new Error(`Failed to compute SHA2-256 hash: ${error.message}`);
  }
}

/**
 * Generates a multihash identifier for an artifact.
 * Combines the SHA2-256 hash with a multihash prefix for future-proofing.
 * Currently uses a simplified prefix '12' (SHA2-256) + length '20' as per multihash spec.
 * 
 * @param {string|Buffer} content - The content to hash.
 * @returns {string} - The multihash string (hex format with prefix).
 * @throws {Error} - If hash computation fails.
 */
function generateMultihash(content) {
  try {
    const hash = computeSHA256(content);
    // Multihash prefix for SHA2-256 (0x12) and length (0x20 = 32 bytes)
    const prefix = '1220';
    return `${prefix}${hash}`;
  } catch (error) {
    throw new Error(`Failed to generate multihash: ${error.message}`);
  }
}

/**
 * Hashes an artifact for storage, ensuring backward compatibility.
 * This function is designed to be called during artifact write operations.
 * If an existing ID is provided and matches a legacy format, it will be preserved.
 * Otherwise, a new multihash-based ID is generated.
 * 
 * @param {string|Buffer} content - The artifact content to hash.
 * @param {string} [existingId] - Optional existing artifact ID for compatibility.
 * @returns {string} - The artifact ID (new multihash or preserved existing ID).
 * @throws {Error} - If content is invalid or processing fails.
 */
function hashArtifact(content, existingId = null) {
  try {
    // Check for existing ID and validate if it matches legacy format
    if (existingId) {
      // Placeholder for legacy ID format check (e.g., non-multihash IDs)
      // For now, assume any existing ID is valid and return it for compatibility
      return existingId;
    }
    // No existing ID, generate a new multihash
    return generateMultihash(content);
  } catch (error) {
    throw new Error(`Failed to hash artifact: ${error.message}`);
  }
}

module.exports = {
  computeSHA256,
  generateMultihash,
  hashArtifact
};
