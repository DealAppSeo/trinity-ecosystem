import time
import random
from datetime import datetime
from typing import Optional, Dict, Any
import logging

class TaskFailurePreventionSystem:
    """Antifragile system to prevent task failures"""
    
    def __init__(self):
        self.failure_history = []
        self.circuit_breakers = {}
        self.retry_config = {
            'max_retries': 3,
            'backoff_factor': 2,
            'initial_delay': 1
        }
        
    def execute_with_resilience(self, task_func, task_id: str, *args, **kwargs):
        """Execute task with built-in failure prevention"""
        
        # Check circuit breaker
        if self._circuit_open(task_id):
            return self._handle_circuit_open(task_id)
        
        # Execute with retry logic
        for attempt in range(self.retry_config['max_retries'] + 1):
            try:
                result = task_func(*args, **kwargs)
                self._record_success(task_id)
                return result
                
            except Exception as e:
                error_type = type(e).__name__
                self._record_failure(task_id, error_type, attempt)
                
                if attempt == self.retry_config['max_retries']:
                    self._trip_circuit(task_id)
                    raise
                
                # Exponential backoff
                delay = self.retry_config['initial_delay'] * \
                       (self.retry_config['backoff_factor'] ** attempt)
                time.sleep(delay)
    
    def _circuit_open(self, task_id: str) -> bool:
        """Check if circuit breaker is open for this task"""
        if task_id not in self.circuit_breakers:
            return False
        
        breaker = self.circuit_breakers[task_id]
        if breaker['state'] == 'OPEN':
            # Check if we should attempt reset
            reset_time = breaker['tripped_at'] + timedelta(minutes=5)
            if datetime.now() > reset_time:
                breaker['state'] = 'HALF_OPEN'
                breaker['test_attempts'] = 0
            else:
                return True
        return False
    
    def _trip_circuit(self, task_id: str):
        """Trip circuit breaker for failed task"""
        self.circuit_breakers[task_id] = {
            'state': 'OPEN',
            'tripped_at': datetime.now(),
            'failure_count': 0
        }
        logging.warning(f"Circuit breaker tripped for task: {task_id}")
    
    def _record_failure(self, task_id: str, error_type: str, attempt: int):
        """Record failure for analysis"""
        failure_record = {
            'task_id': task_id,
            'error_type': error_type,
            'attempt': attempt,
            'timestamp': datetime.now().isoformat(),
            'circuit_state': self.circuit_breakers.get(task_id, {}).get('state', 'CLOSED')
        }
        self.failure_history.append(failure_record)
        
        # Update circuit breaker failure count
        if task_id in self.circuit_breakers:
            self.circuit_breakers[task_id]['failure_count'] += 1
    
    def _record_success(self, task_id: str):
        """Record successful execution"""
        if task_id in self.circuit_breakers:
            breaker = self.circuit_breakers[task_id]
            if breaker['state'] == 'HALF_OPEN':
                breaker['test_attempts'] += 1
                if breaker['test_attempts'] >= 3:
                    breaker['state'] = 'CLOSED'
                    logging.info(f"Circuit breaker reset for task: {task_id}")
    
    def _handle_circuit_open(self, task_id: str):
        """Handle requests when circuit is open"""
        return {
            'status': 'circuit_open',
            'task_id': task_id,
            'message': 'Task temporarily disabled due to repeated failures',
            'suggested_action': 'Wait 5 minutes or implement fallback logic',
            'timestamp': datetime.now().isoformat()
        }
    
    def get_failure_analytics(self) -> Dict[str, Any]:
        """Generate analytics from failure history"""
        if not self.failure_history:
            return {'total_failures': 0}
        
        error_counts = {}
        for failure in self.failure_history:
            error_type = failure['error_type']
            error_counts[error_type] = error_counts.get(error_type, 0) + 1
        
        return {
            'total_failures': len(self.failure_history),
            'error_distribution': error_counts,
            'recent_failures': self.failure_history[-10:],  # Last 10 failures
            'active_circuit_breakers': len([cb for cb in self.circuit_breakers.values() 
                                           if cb['state'] != 'CLOSED'])
        }

# save_artifact: failure_monitoring_dashboard.py
from dataclasses import dataclass
from typing import List, Dict
import asyncio
from datetime import datetime, timedelta

@dataclass
class FailureAlert:
    task_id: str
    error_type: str
    severity: str
    timestamp: datetime
    recommendation: str

class FailureMonitoringDashboard:
    """Real-time monitoring for task failures"""
    
    def __init__(self):
        self.failure_thresholds = {
            'critical': 10,  # Failures per hour
            'warning': 5,    # Failures per hour
            'info': 2        # Failures per hour
        }
        self.alerts = []
        self.failure_window = []
        
    def record_failure(self, task_id: str, error_type: str):
        """Record a failure and check thresholds"""
        timestamp = datetime.now()
        self.failure_window.append({
            'task_id': task_id,
            'error_type': error_type,
            'timestamp': timestamp
        })
        
        # Clean old failures (keep last hour)
        one_hour_ago = timestamp - timedelta(hours=1)
        self.failure_window = [
            f for f in self.failure_window 
            if f['timestamp'] > one_hour_ago
        ]
        
        # Check thresholds
        self._check_thresholds(task_id, error_type)
    
    def _check_thresholds(self, task_id: str, error_type: str):
        """Check if failure thresholds are exceeded"""
        recent_failures = [
            f for f in self.failure_window 
            if f['task_id'] == task_id and f['error_type'] == error_type
        ]
        
        failure_count = len(recent_failures)
        
        if failure_count >= self.failure_thresholds['critical']:
            self._create_alert(
                task_id, error_type, 'critical',
                f"Critical: {failure_count} failures in last hour for {task_id}. "
                f"Immediate intervention required."
            )
        elif failure_count >= self.failure_thresholds['warning']:
            self._create_alert(
                task_id, error_type, 'warning',
                f"Warning: {failure_count} failures in last hour for {task_id}. "
                f"Monitor closely."
            )
        elif failure_count >= self.failure_thresholds['info']:
            self._create_alert(
                task_id, error_type, 'info',
                f"Info: {failure_count} failures in last hour for {task_id}. "
                f"Normal operation but worth monitoring."
            )
    
    def _create_alert(self, task_id: str, error_type: str, 
                     severity: str, message: str):
        """Create and store an alert"""
        alert = FailureAlert(
            task_id=task_id,
            error_type=error_type,
            severity=severity,
            timestamp=datetime.now(),
            recommendation=message
        )
        self.alerts.append(alert)
        
        # In production, this would send to monitoring system
        print(f"[{severity.upper()}] {message}")
    
    def get_dashboard_data(self) -> Dict:
        """Get current dashboard state"""
        now = datetime.now()
        hour_ago = now - timedelta(hours=1)
        
        recent_failures = [
            f for f in self.failure_window 
            if f['timestamp'] > hour_ago
        ]
        
        # Group by task and error type
        failure_summary = {}
        for failure in recent_failures:
            key = f"{failure['task_id']}:{failure['error_type']}"
            failure_summary[key] = failure_summary.get(key, 0) + 1
        
        return {
            'total_failures_last_hour': len(recent_failures),
            'failure_summary': failure_summary,
            'active_alerts': len([a for a in self.alerts 
                                 if a.timestamp > hour_ago]),
            'alerts_by_severity': self._count_alerts_by_severity()
        }
    
    def _count_alerts_by_severity(self) -> Dict[str, int]:
        """Count alerts by severity level"""
        counts = {'critical': 0, 'warning': 0, 'info': 0}
        hour_ago = datetime.now() - timedelta(hours=1)
        
        for alert in self.alerts:
            if alert.timestamp > hour_ago:
                counts[alert.severity] += 1
        
        return counts
