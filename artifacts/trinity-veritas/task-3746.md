circuit_breaker_code = '''
class AntifragileCircuitBreaker:
    """Circuit breaker pattern implementation for task_failed prevention"""
    
    def __init__(self, failure_threshold=5, recovery_timeout=60):
        self.failure_threshold = failure_threshold
        self.recovery_timeout = recovery_timeout
        self.failure_count = 0
        self.last_failure_time = None
        self.state = "CLOSED"  # CLOSED, OPEN, HALF_OPEN
        
    def execute_task(self, task_func, *args, **kwargs):
        """Execute task with circuit breaker protection"""
        
        if self.state == "OPEN":
            if self._can_attempt_recovery():
                self.state = "HALF_OPEN"
            else:
                raise CircuitBreakerOpenError(
                    f"Circuit breaker is OPEN. Next recovery attempt in {self._time_to_recovery()} seconds"
                )
        
        try:
            result = task_func(*args, **kwargs)
            self._on_success()
            return result
            
        except Exception as e:
            self._on_failure(e)
            raise
    
    def _on_success(self):
        """Handle successful execution"""
        if self.state == "HALF_OPEN":
            self.state = "CLOSED"
        self.failure_count = 0
    
    def _on_failure(self, error):
        """Handle failed execution"""
        self.failure_count += 1
        self.last_failure_time = datetime.now()
        
        if self.failure_count >= self.failure_threshold:
            self.state = "OPEN"
        
        # Log failure with context for analysis
        self._log_failure_context(error)
    
    def _can_attempt_recovery(self):
        """Check if recovery attempt is allowed"""
        if not self.last_failure_time:
            return True
        
        time_since_failure = (datetime.now() - self.last_failure_time).total_seconds()
        return time_since_failure >= self.recovery_timeout
    
    def _time_to_recovery(self):
        """Calculate time until next recovery attempt"""
        if not self.last_failure_time:
            return 0
        
        elapsed = (datetime.now() - self.last_failure_time).total_seconds()
        return max(0, self.recovery_timeout - elapsed)
    
    def _log_failure_context(self, error):
        """Log comprehensive failure context for analysis"""
        failure_context = {
            "timestamp": datetime.now().isoformat(),
            "failure_count": self.failure_count,
            "state": self.state,
            "error_type": type(error).__name__,
            "error_message": str(error),
            "circuit_breaker_threshold": self.failure_threshold
        }
        # In production, log to centralized logging system
        print(f"[FAILURE_CONTEXT] {json.dumps(failure_context)}")


class ExponentialBackoffRetry:
    """Intelligent retry logic with exponential backoff"""
    
    def __init__(self, max_retries=5, base_delay=1, max_delay=60):
        self.max_retries = max_retries
        self.base_delay = base_delay
        self.max_delay = max_delay
    
    def execute_with_retry(self, task_func, *args, **kwargs):
        """Execute task with exponential backoff retry"""
        
        last_exception = None
        
        for attempt in range(self.max_retries + 1):
            try:
                if attempt > 0:
                    delay = min(self.base_delay * (2 ** (attempt - 1)), self.max_delay)
                    print(f"Retry attempt {attempt}/{self.max_retries} after {delay}s delay")
                    time.sleep(delay)
                
                return task_func(*args, **kwargs)
                
            except Exception as e:
                last_exception = e
                print(f"Attempt {attempt} failed: {type(e).__name__}: {str(e)}")
                
                # Don't retry on certain permanent errors
                if self._is_permanent_error(e):
                    break
        
        raise MaxRetriesExceededError(
            f"Failed after {self.max_retries} retries. Last error: {str(last_exception)}"
        ) from last_exception
    
    def _is_permanent_error(self, error):
        """Determine if error is permanent (should not retry)"""
        permanent_error_indicators = [
            "permission denied",
            "not found",
            "invalid",
            "syntax error",
            "validation failed"
        ]
        
        error_msg = str(error).lower()
        return any(indicator in error_msg for indicator in permanent_error_indicators)
'''
