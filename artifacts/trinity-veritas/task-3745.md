class CircuitBreaker:
    """Implements circuit breaker pattern for task execution"""
    
    def __init__(self, name: str, failure_threshold: int = 5, reset_timeout: int = 60):
        self.name = name
        self.failure_threshold = failure_threshold
        self.reset_timeout = reset_timeout
        self.failure_count = 0
        self.state = 'CLOSED'  # CLOSED, OPEN, HALF_OPEN
        self.last_failure_time = None
        self.last_state_change = datetime.now()
        self.lock = threading.Lock()
        
    def execute(self, task_func, *args, **kwargs):
        """Execute task with circuit breaker protection"""
        with self.lock:
            current_time = time.time()
            
            # Check if circuit is open
            if self.state == 'OPEN':
                if self.last_failure_time and \
                   (current_time - self.last_failure_time) > self.reset_timeout:
                    self.state = 'HALF_OPEN'
                    logger.info(f"Circuit {self.name}: OPEN → HALF_OPEN")
                else:
                    raise CircuitBreakerOpenError(
                        f"Circuit breaker {self.name} is OPEN. "
                        f"Last failure: {self.last_failure_time}"
                    )
            
            try:
                result = task_func(*args, **kwargs)
                
                # If successful in HALF_OPEN state, close the circuit
                if self.state == 'HALF_OPEN':
                    self.state = 'CLOSED'
                    self.failure_count = 0
                    logger.info(f"Circuit {self.name}: HALF_OPEN → CLOSED")
                
                return result
                
            except Exception as e:
                self._record_failure()
                raise

        """Record a failure and update circuit state"""

class ExponentialBackoffRetry:
    """Implements retry logic with exponential backoff"""
    
    def __init__(self, max_retries: int = 3, base_delay: float = 1.0):
        self.max_retries = max_retries
        self.base_delay = base_delay
        
    def execute(self, task_func, *args, **kwargs):
        """Execute task with exponential backoff retry"""
        last_exception = None
        
        for attempt in range(self.max_retries + 1):
            try:
                if attempt > 0:
                    logger.info(f"Retry attempt {attempt}/{self.max_retries}")
                
                return task_func(*args, **kwargs)
                
            except Exception as e:
                last_exception = e
                
                # Don't retry on certain errors
                if self._should_not_retry(e):
                    break
                
                if attempt < self.max_retries:
                    delay = self.base_delay * (2 ** attempt)
                    logger.info(f"Attempt {attempt + 1} failed: {str(e)[:100]}... "
                              f"Retrying in {delay:.1f}s")
                    time.sleep(delay)
        
        raise last_exception or Exception("Task failed after all retries")
    
    def _should_not_retry(self, error: Exception) -> bool:
        """Determine if error should not be retried"""
        error_str = str(error).lower()
        non_retryable = [
            'syntax', 'validation', 'permission', 'configuration',
            'not found', 'invalid', 'corrupt'
        ]
        
        for phrase in non_retryable:
            if phrase in error_str:
                return True
        return False
