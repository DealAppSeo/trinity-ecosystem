class FailurePatternAnalyzer:

    def __init__(self):

    def log_failure(self, task_id: str, error_type: str, 
                   context: Dict, timestamp: datetime = None):
        """Log a failure with context for pattern analysis"""
        if timestamp is None:
            timestamp = datetime.now()
            
        failure_record = {
            'task_id': task_id,
            'error_type': error_type,
            'timestamp': timestamp.isoformat(),
            'context': context,
            'hash': self._generate_context_hash(context)
        }
        
        self.failure_log.append(failure_record)
        self._analyze_patterns()
        
    def _generate_context_hash(self, context: Dict) -> str:
        """Generate hash for failure context to identify duplicates"""
        context_str = json.dumps(context, sort_keys=True)
        return hashlib.md5(context_str.encode()).hexdigest()[:8]
    
    def _analyze_patterns(self):
        """Analyze failure patterns and update prevention rules"""
        # Group by error type and context hash
        recent_failures = [f for f in self.failure_log 
                          if datetime.fromisoformat(f['timestamp']) > 
                          datetime.now() - timedelta(hours=24)]
        
        patterns = {}
        for failure in recent_failures:
            key = (failure['error_type'], failure['hash'])
            if key not in patterns:
                patterns[key] = []
            patterns[key].append(failure)
        
        # Identify high-frequency patterns
        for (error_type, context_hash), failures in patterns.items():
            if len(failures) >= 3:  # Threshold for pattern detection
                pattern_id = f"{error_type}_{context_hash}"
                self.patterns[pattern_id] = {
                    'error_type': error_type,

class TaskExecutionGuard:
    """Wraps task execution with failure prevention"""
    
    def __init__(self, analyzer: FailurePatternAnalyzer):
        self.analyzer = analyzer
        self.execution_stats = {
            'total_tasks': 0,
            'successful': 0,
            'prevented': 0,
            'failed': 0
        }
    
    def execute_with_guard(self, task_func, task_id: str, 
                          context: Dict, *args, **kwargs) -> Any:
        """Execute task with failure prevention guard"""
        self.execution_stats['total_tasks'] += 1
        
        # Check prevention rules
        prevention_check = self.analyzer.should_prevent('task_failed', context)
        if prevention_check['prevent']:
            self.execution_stats['prevented'] += 1
            raise Exception(
                f"Task prevented: {prevention_check['reason']}\n"
                f"Alternative: {prevention_check['alternative']}"
            )
        
        try:
            result = task_func(*args, **kwargs)
            self.execution_stats['successful'] += 1
            return result
            
        except Exception as e:
            self.execution_stats['failed'] += 1
            
            # Log the failure
            self.analyzer.log_failure(
                task_id=task_id,
                error_type='task_failed',
                context={
                    'error': str(e),
                    'args': str(args),
                    'kwargs': str(kwargs),
                    'context': context,
                    'timestamp': datetime.now().isoformat()
                }
            )
            
            # Re-raise with additional context
            raise Exception(
                f"Task failed: {str(e)}\n"
                f"Failure logged for pattern analysis (ID: {task_id})"
            )
    
    def get_stats(self) -> Dict:
        """Get execution statistics"""
        return self.execution_stats
