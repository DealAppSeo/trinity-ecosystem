"""
Enhanced Error Context Wrapper for task_failed prevention
Implements TRUE virtue: Never fabricate, log everything
"""

import traceback
import logging
from datetime import datetime
from typing import Any, Callable, Dict, Optional
import json

logging.basicConfig(level=logging.INFO)
logger = logging.getLogger(__name__)

class TaskFailedError(Exception):
    """Enhanced task failure exception with full context"""
    def __init__(self, error_context: Dict[str, Any]):
        self.error_context = error_context
        super().__init__(f"Task failed: {error_context.get('error_type', 'Unknown')}")

class TaskContextWrapper:
    """Wraps task execution with comprehensive error context"""
    
    def __init__(self, system_name: str = "trinity-veritas"):
        self.system_name = system_name
        self.task_registry = {}
        
    def execute(self, task_id: str, task_func: Callable, 
                *args, **kwargs) -> Any:
        """
        Execute a task with full error context capture
        
        Args:
            task_id: Unique task identifier
            task_func: Function to execute
            *args, **kwargs: Task function arguments
            
        Returns:
            Task result if successful
            
        Raises:
            TaskFailedError: With complete error context
        """
        start_time = datetime.now()
        
        try:
            # Execute the task
            result = task_func(*args, **kwargs)
            
            # Log success
            success_context = {
                'task_id': task_id,
                'status': 'SUCCESS',
                'start_time': start_time.isoformat(),
                'end_time': datetime.now().isoformat(),
                'duration_seconds': (datetime.now() - start_time).total_seconds(),
                'function': task_func.__name__,
                'system': self.system_name
            }
            
            self._log_success(success_context)
            self.task_registry[task_id] = {
                'last_status': 'SUCCESS',
                'last_execution': datetime.now(),
                'failure_count': 0
            }
            
            return result
            
        except Exception as e:
            # Capture complete error context
            error_context = {
                'task_id': task_id,
                'status': 'FAILED',
                'start_time': start_time.isoformat(),
                'end_time': datetime.now().isoformat(),
                'duration_seconds': (datetime.now() - start_time).total_seconds(),
                'function': task_func.__name__,
                'args_preview': str(args)[:200] if args else 'None',
                'kwargs_keys': list(kwargs.keys()) if kwargs else [],
                'error_type': type(e).__name__,
                'error_message': str(e),
                'stack_trace': traceback.format_exc(),
                'system': self.system_name,
                'failure_sequence': self._get_failure_sequence(task_id)
            }
            
            self._log_failure(error_context)
            self._update_failure_stats(task_id)
            
            # Raise enhanced error
            raise TaskFailedError(error_context) from e
    
    def _log_success(self, context: Dict[str, Any]):
        """Log successful task execution"""
        log_entry = {
            **context,
            'log_level': 'INFO',
            'event': 'TASK_SUCCESS'
        }
        logger.info(json.dumps(log_entry, default=str))
        
        # Store in task registry
        if context['task_id'] not in self.task_registry:
            self.task_registry[context['task_id']] = {
                'success_count': 0,
                'failure_count': 0
            }
        self.task_registry[context['task_id']]['success_count'] = \
            self.task_registry[context['task_id']].get('success_count', 0) + 1
    
    def _log_failure(self, context: Dict[str, Any]):
        """Log task failure with full context"""
        log_entry = {
            **context,
            'log_level': 'ERROR',
            'event': 'TASK_FAILED'
        }
        logger.error(json.dumps(log_entry, default=str))
        
        # Store for analysis
        self._store_failure_for_analysis(context)
    
    def _get_failure_sequence(self, task_id: str) -> int:
        """Get sequential failure count for this task"""
        if task_id not in self.task_registry:
            return 1
        return self.task_registry[task_id].get('failure_sequence', 0) + 1
    
    def _update_failure_stats(self, task_id: str):
        """Update failure statistics for task"""
        if task_id not in self.task_registry:
            self.task_registry[task_id] = {
                'failure_count': 0,
                'success_count': 0,
                'last_failure': datetime.now()
            }
        
        self.task_registry[task_id]['failure_count'] = \
            self.task_registry[task_id].get('failure_count', 0) + 1
        self.task_registry[task_id]['last_failure'] = datetime.now()
        self.task_registry[task_id]['failure_sequence'] = \
            self.task_registry[task_id].get('failure_sequence', 0) + 1
        
        # Reset success sequence
        self.task_registry[task_id]['success_sequence'] = 0
    
    def _store_failure_for_analysis(self, context: Dict[str, Any]):
        """Store failure context for pattern analysis"""
        # This would typically write to a database or file
        # For now, we'll append to a log file
        try:
            with open('task_failures.log', 'a') as f:
                f.write(json.dumps(context, default=str) + '\n')
        except:
            pass  # Don't fail on logging failure
    
    def get_task_stats(self, task_id: str) -> Optional[Dict[str, Any]]:
        """Get statistics for a specific task"""
        return self.task_registry.get(task_id)
    
    def get_all_stats(self) -> Dict[str, Dict[str, Any]]:
        """Get statistics for all tasks"""
        return self.task_registry.copy()

# Global instance for easy import
task_wrapper = TaskContextWrapper()

# Decorator version for convenience
def task_with_context(task_id: str):
    """Decorator to wrap functions with error context"""
    def decorator(func):
        def wrapper(*args, **kwargs):
            return task_wrapper.execute(task_id, func, *args, **kwargs)
        return wrapper
    return decorator

import time
from datetime import datetime, timedelta
from typing import Any, Callable, Optional, Dict
import logging
from enum import Enum

logger = logging.getLogger(__name__)

class CircuitState(Enum):
    CLOSED = "CLOSED"      # Normal operation
    OPEN = "OPEN"          # Fail-fast, no requests
    HALF_OPEN = "HALF_OPEN" # Testing recovery

class CircuitOpenError(Exception):
    """Exception raised when circuit breaker is open"""
    pass

class CircuitBreaker:
    """
    Implements circuit breaker pattern for fault tolerance
    
    Principles:
    1. Fail fast when system is unhealthy
    2. Allow periodic testing for recovery
    3. Automatic reset after successful operations
    """
    
    def __init__(self, 
                 name: str,
                 failure_threshold: int = 5,
                 reset_timeout: int = 60,
                 half_open_timeout: int = 10,
                 half_open_max_attempts: int = 3):
        """
        Initialize circuit breaker
        
        Args:
            name: Circuit identifier
            failure_threshold: Consecutive failures before opening
            reset_timeout: Seconds before attempting half-open
            half_open_timeout: Time window for half-open testing
            half_open_max_attempts: Max attempts in half-open state
        """
        self.name = name
        self.failure_threshold = failure_threshold
        self.reset_timeout = reset_timeout
        self.half_open_timeout = half_open_timeout
        self.half_open_max_attempts = half_open_max_attempts
        
        # State
        self.state = CircuitState.CLOSED
        self.failure_count = 0
        self.last_failure_time = None
        self.last_success_time = None
        self.half_open_attempts = 0
        self.half_open_start = None
        
        # Statistics
        self.total_requests = 0
        self.total_failures = 0
        self.total_successes = 0
        self.state_changes = []
        
        logger.info(f"CircuitBreaker '{name}' initialized: CLOSED state")
    
    def execute(self, operation: Callable, *args, **kwargs) -> Any:
        """
        Execute operation with circuit breaker protection
        
        Args:
            operation: Function to execute
            *args, **kwargs: Operation arguments
            
        Returns:
            Operation result
            
        Raises:
            CircuitOpenError: If circuit is open
            Exception: Any exception from the operation
        """
        self.total_requests += 1
        
        # Check circuit state
        self._update_state()
        
        if self.state == CircuitState.OPEN:
            logger.warning(f"Circuit '{self.name}' OPEN - failing fast")
            raise CircuitOpenError(
                f"Circuit '{self.name}' is OPEN. "
                f"Last failure: {self.last_failure_time}"
            )
        
        # Execute operation
        try:
            result = operation(*args, **kwargs)
            self._on_success()
            return result
            
        except Exception as e:
            self._on_failure(e)
            raise
    
    def _update_state(self):
        """Update circuit state based on timing and thresholds"""
        current_time = datetime.now()
        
        # State transition logic
        if self.state == CircuitState.OPEN:
            # Check if reset timeout has passed
            if (self.last_failure_time and 
                (current_time - self.last_failure_time).total_seconds() > self.reset_timeout):
                self._transition_to(CircuitState.HALF_OPEN)
                self.half_open_start = current_time
                self.half_open_attempts = 0
                
        elif self.state == CircuitState.HALF_OPEN:
            # Check if half-open timeout exceeded
            if (self.half_open_start and 
                (current_time - self.half_open_start).total_seconds() > self.half_open_timeout):
                self._transition_to(CircuitState.OPEN)
            
            # Check if max attempts exceeded
            if self.half_open_attempts >= self.half_open_max_attempts:
                self._transition_to(CircuitState.OPEN)
    
    def _on_success(self):
        """Handle successful operation"""
        self.total_successes += 1
        self.last_success_time = datetime.now()
        
        if self.state == CircuitState.HALF_OPEN:
            # Success in half-open state - reset to CLOSED
            self._transition_to(CircuitState.CLOSED)
            self.failure_count = 0
            self.half_open_attempts = 0
            self.half_open_start = None
            
        elif self.state == CircuitState.CLOSED:
            # Reset failure count on consecutive successes
            if self.failure_count > 0:
                self.failure_count = max(0, self.failure_count - 1)
    
    def _on_failure(self, error: Exception):
        """Handle failed operation"""
        self.total_failures += 1
        self.failure_count += 1
        self.last_failure_time = datetime.now()
        
        if self.state == CircuitState.HALF_OPEN:
            self.half_open_attempts += 1
            logger.warning(
                f"Circuit '{self.name}' HALF_OPEN attempt {self.half_open_attempts} failed"
            )

        if (self.state == CircuitState.CLOSED and 
            self.failure_count >= self.failure_threshold):
            self._transition_to(CircuitState.OPEN)
            
        elif self.state == CircuitState.HALF_OPEN:
            # Failure in half-open state
            logger.warning(f"Circuit '{self.name}' HALF_OPEN test failed")
