"""
Circuit Breaker Implementation for Fast Execution Tasks
Prevents cascade failures by stopping execution after threshold
"""

import time
import threading
from enum import Enum
from typing import Callable, Any
import logging

logger = logging.getLogger(__name__)

class CircuitState(Enum):
    CLOSED = "CLOSED"  # Normal operation
    OPEN = "OPEN"      # Circuit open, fast fail
    HALF_OPEN = "HALF_OPEN"  # Testing recovery

class CircuitBreaker:
    def __init__(self, 
                 name: str,
                 failure_threshold: int = 5,
                 reset_timeout: int = 60,
                 half_open_max_attempts: int = 3):
        self.name = name
        self.failure_threshold = failure_threshold
        self.reset_timeout = reset_timeout
        self.half_open_max_attempts = half_open_max_attempts
        
        self.state = CircuitState.CLOSED
        self.failure_count = 0
        self.last_failure_time = None
        self.half_open_attempts = 0
        self.lock = threading.RLock()
        
        # Statistics
        self.total_failures = 0
        self.total_requests = 0
        self.last_reset_time = time.time()
    
    def execute(self, operation: Callable[[], Any]) -> Any:
        """Execute operation with circuit breaker protection"""
        with self.lock:
            self.total_requests += 1
            
            if self.state == CircuitState.OPEN:
                # Check if reset timeout has passed
                if (time.time() - self.last_failure_time) > self.reset_timeout:
                    self.state = CircuitState.HALF_OPEN
                    self.half_open_attempts = 0
                    logger.info(f"Circuit {self.name} moving to HALF_OPEN state")
                else:
                    raise CircuitBreakerOpenError(
                        f"Circuit {self.name} is OPEN. "
                        f"Next test in {self.reset_timeout - (time.time() - self.last_failure_time):.1f}s"
                    )
            
            if self.state == CircuitState.HALF_OPEN:
                if self.half_open_attempts >= self.half_open_max_attempts:
                    # Too many attempts, stay open longer
                    self.state = CircuitState.OPEN
                    self.last_failure_time = time.time()
                    raise CircuitBreakerOpenError(
                        f"Circuit {self.name} failed recovery attempts"
                    )
                self.half_open_attempts += 1
            
            try:
                result = operation()
                
                # Success - reset circuit if needed
                if self.state == CircuitState.HALF_OPEN:
                    self.reset()
                    logger.info(f"Circuit {self.name} reset to CLOSED after successful execution")
                
                self.failure_count = 0
                return result
                
            except Exception as e:
                self._record_failure()
                logger.warning(f"Circuit {self.name} failure: {str(e)}")
                raise e
    
    def _record_failure(self):
        """Record a failure and update circuit state"""
        with self.lock:
            self.failure_count += 1
            self.total_failures += 1
            self.last_failure_time = time.time()
            
            if self.state == CircuitState.HALF_OPEN:
                # Failed during recovery attempt
                self.state = CircuitState.OPEN
                logger.warning(f"Circuit {self.name} failed recovery, returning to OPEN")
            
            elif self.failure_count >= self.failure_threshold:
                # Threshold exceeded, open circuit
                self.state = CircuitState.OPEN
                logger.error(
                    f"Circuit {self.name} opened after {self.failure_count} failures. "
                    f"Will reset in {self.reset_timeout}s"
                )
    
    def reset(self):
        """Manually reset the circuit breaker"""
        with self.lock:
            self.state = CircuitState.CLOSED
            self.failure_count = 0
            self.half_open_attempts = 0
            self.last_reset_time = time.time()
            logger.info(f"Circuit {self.name} manually reset")
    
    def get_stats(self) -> dict:
        """Get circuit breaker statistics"""
        with self.lock:
            return {
                "name": self.name,
                "state": self.state.value,
                "failure_count": self.failure_count,
                "total_failures": self.total_failures,
                "total_requests": self.total_requests,
                "failure_rate": (self.total_failures / self.total_requests * 100) 
                               if self.total_requests > 0 else 0,
                "last_failure_time": self.last_failure_time,
                "last_reset_time": self.last_reset_time,
                "is_open": self.state == CircuitState.OPEN
            }

class CircuitBreakerOpenError(Exception):
    """Exception raised when circuit breaker is open"""
    pass

# Global registry for monitoring
class CircuitBreakerRegistry:
    _breakers = {}

    def register(cls, name: str, **kwargs) -> CircuitBreaker:
        if name not in cls._breakers:
            cls._breakers[name] = CircuitBreaker(name, **kwargs)
        return cls._breakers[name]
    
    @classmethod
    def get_all_stats(cls) -> dict:
        return {name: breaker.get_stats() for name, breaker in cls._breakers.items()}
