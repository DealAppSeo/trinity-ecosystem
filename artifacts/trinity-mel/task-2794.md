```python
# Example code for implementing a simple GNN
import torch
import torch.nn as nn
import torch_geometric.nn as pyg_nn

class SimpleGNN(nn.Module):
    def __init__(self):
        super(SimpleGNN, self).__init__()
        self.conv1 = pyg_nn.GraphConv(16, 32)
        self.conv2 = pyg_nn.GraphConv(32, 64)

    def forward(self, data):
        x, edge_index = data.x, data.edge_index

        x = torch.relu(self.conv1(x, edge_index))
        x = self.conv2(x, edge_index)
        return x

# Saving the GNN model as an artifact

torch.save(gnn_model.state_dict(), "gnn_model.pth")
save_artifact("gnn_model.pth", "GNN model parameters")
```
