class Agent:

        self.id = id
        self.neighbors = []
        self.target_location = None

        self.neighbors = [agent for agent in agents if agent.id != self.id and distance(self, agent) < 10]

        self.target_location = target_location

    def move(self):
        if distance(self, self.target_location) < 5:
            move_towards(self, self.target_location)
        elif self.neighbors:
            average_location = average_location(self.neighbors)
            move_towards(self, average_location)
        else:
            move_towards(self, self.target_location)

def distance(agent1, agent2):
    return ((agent1.x - agent2.x) ** 2 + (agent1.y - agent2.y) ** 2) ** 0.5

def average_location(agents):
    x_sum = sum(agent.x for agent in agents)
    y_sum = sum(agent.y for agent in agents)
    return (x_sum / len(agents), y_sum / len(agents))

def move_towards(agent, target):
    # Simple movement algorithm
    agent.x += (target[0] - agent.x) * 0.1
    agent.y += (target[1] - agent.y) * 0.1
