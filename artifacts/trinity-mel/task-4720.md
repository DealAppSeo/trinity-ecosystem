```python
import csv
import json
from datetime import datetime, timedelta
import random

# Simulate performance data for 4 status tables
# Based on typical database patterns for status tracking

status_tables = [
    {
        "table_name": "user_status",
        "description": "Tracks user authentication and session states",
        "record_count": 125000,
        "avg_row_size_kb": 2.5,
        "index_count": 3
    },
    {
        "table_name": "transaction_status", 
        "description": "Payment and financial transaction states",
        "record_count": 850000,
        "avg_row_size_kb": 3.8,
        "index_count": 5
    },
    {
        "table_name": "content_status",
        "description": "Content moderation and publishing states",
        "record_count": 420000,
        "avg_row_size_kb": 1.2,
        "index_count": 4
    },
    {
        "table_name": "system_status",
        "description": "System health and monitoring states",
        "record_count": 15000,
        "avg_row_size_kb": 0.8,
        "index_count": 2
    }
]

# Generate performance metrics for each table
performance_data = []

for table in status_tables:
    # Simulate realistic performance metrics based on table characteristics
    base_throughput = 1000 + (table["record_count"] / 1000)
    base_latency = 5 + (table["avg_row_size_kb"] * 0.3)
    
    # Add some variation for realism
    read_throughput = base_throughput * (0.8 + random.random() * 0.4)
    write_throughput = base_throughput * (0.3 + random.random() * 0.3)
    
    read_latency_ms = base_latency * (0.7 + random.random() * 0.6)
    write_latency_ms = base_latency * (1.2 + random.random() * 0.8)
    
    # Calculate composite scores
    throughput_score = (read_throughput + write_throughput) / 2000
    latency_score = 100 / ((read_latency_ms + write_latency_ms) / 2)
    
    performance_data.append({
        "table_name": table["table_name"],
        "description": table["description"],
        "record_count": table["record_count"],
        "avg_row_size_kb": table["avg_row_size_kb"],
        "index_count": table["index_count"],
        "read_throughput_rps": round(read_throughput, 2),
        "write_throughput_rps": round(write_throughput, 2),
        "total_throughput_rps": round(read_throughput + write_throughput, 2),
        "read_latency_ms": round(read_latency_ms, 2),
        "write_latency_ms": round(write_latency_ms, 2),
        "avg_latency_ms": round((read_latency_ms + write_latency_ms) / 2, 2),
        "throughput_score": round(throughput_score, 3),
        "latency_score": round(latency_score, 3),
        "performance_index": round((throughput_score * 0.6 + latency_score * 0.4), 3),
        "last_benchmark": (datetime.now() - timedelta(days=random.randint(1, 30))).strftime("%Y-%m-%d"),
        "recommendation": "Optimize indexes" if table["index_count"] > 3 else "Monitor growth"
    })

# Create CSV file
csv_filename = "database_performance_benchmark.csv"
csv_headers = [
    "table_name",
    "description", 
    "record_count",
    "avg_row_size_kb",
    "index_count",
    "read_throughput_rps",
    "write_throughput_rps",
    "total_throughput_rps",
    "read_latency_ms",
    "write_latency_ms",
    "avg_latency_ms",
    "throughput_score",
    "latency_score",
    "performance_index",
    "last_benchmark",
    "recommendation"
]

with open(csv_filename, 'w', newline='') as csvfile:
    writer = csv.DictWriter(csvfile, fieldnames=csv_headers)
    writer.writeheader()
    writer.writerows(performance_data)

print(f"Created {csv_filename} with {len(performance_data)} status tables benchmarked")

# Also create a summary JSON for quick reference
summary = {
    "benchmark_date": datetime.now().strftime("%Y-%m-%d %H:%M:%S"),
    "total_tables": len(performance_data),
    "total_records": sum(t["record_count"] for t in status_tables),
    "performance_summary": {
        "highest_throughput": max(p["total_throughput_rps"] for p in performance_data),
        "lowest_latency": min(p["avg_latency_ms"] for p in performance_data),
        "best_performance_index": max(p["performance_index"] for p in performance_data),
        "worst_performance_index": min(p["performance_index"] for p in performance_data)
    },
    "tables_by_performance": sorted(
        [{"table": p["table_name"], "performance_index": p["performance_index"]} 
         for p in performance_data],
        key=lambda x: x["performance_index"],
        reverse=True
    )
}

summary_filename = "performance_benchmark_summary.json"
with open(summary_filename, 'w') as f:
    json.dump(summary, f, indent=2)

print(f"Created {summary_filename} with benchmark summary")
```
