def calculate_code_quality(score_readability, score_maintainability, score_performance):
    # Simplistic approach to calculate overall code quality score
    overall_score = (score_readability + score_maintainability + score_performance) / 3
    return overall_score

def generate_report(score):
    report = f"Code Quality Report:\nOverall Score: {score}\n"
    if score > 0.7:
        report += "The code quality is good. However, there is always room for improvement.\n"
    elif score > 0.4:
        report += "The code quality is fair. Attention should be given to improving readability, maintainability, and performance.\n"
    else:
        report += "The code quality is poor. Significant improvements are needed in terms of readability, maintainability, and performance.\n"
    return report

# Example scores for demonstration
readability_score = 0.8
maintainability_score = 0.7
performance_score = 0.9

overall_score = calculate_code_quality(readability_score, maintainability_score, performance_score)
report = generate_report(overall_score)

# Save the report to a file
with open("code_quality_report.txt", "w") as file:
    file.write(report)

print("Report generated and saved as code_quality_report.txt")
