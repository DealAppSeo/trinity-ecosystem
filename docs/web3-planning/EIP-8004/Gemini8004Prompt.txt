import { openDB, DBSchema, IDBPDatabase } from 'idb';

interface TrinityDB extends DBSchema {
  'repid-scores': {
    key: string;
    value: {
      agentId: string;
      score: number;
      tier: 'Seedling' | 'Sapling' | 'Grove' | 'Forest' | 'Canopy';
      cachedAt: number;
      synced: boolean;
    };
  };
  'pending-updates': {
    key: string;
    value: {
      agentId: string;
      delta: number;
      timestamp: number;
    };
  };
  'localized-virtues': {
    key: string;
    value: {
      virtueKey: string;
      lang: string;
      text: string;
    };
  };
}

let dbInstance: IDBPDatabase<TrinityDB> | null = null;

async function getDB(): Promise<IDBPDatabase<TrinityDB>> {
  if (dbInstance) return dbInstance;
  
  dbInstance = await openDB<TrinityDB>('trinity-cache', 1, {
    upgrade(db) {
      db.createObjectStore('repid-scores', { keyPath: 'agentId' });
      db.createObjectStore('pending-updates', { keyPath: 'agentId' });
      db.createObjectStore('localized-virtues', { keyPath: 'virtueKey' });
    },
  });
  
  return dbInstance;
}

export async function cacheRepID(
  agentId: string, 
  score: number
): Promise<void> {
  const db = await getDB();
  const tier = getTierFromScore(score);
  
  await db.put('repid-scores', {
    agentId,
    score,
    tier,
    cachedAt: Date.now(),
    synced: true,
  });
}

export async function queueRepIDUpdate(
  agentId: string, 
  delta: number
): Promise<void> {
  const db = await getDB();
  
  await db.put('pending-updates', {
    agentId,
    delta,
    timestamp: Date.now(),
  });
}

export async function syncPendingUpdates(): Promise<number> {
  if (!navigator.onLine) return 0;
  
  const db = await getDB();
  const pending = await db.getAll('pending-updates');
  let synced = 0;
  
  for (const update of pending) {
    try {
      // Push to chain via trinity-web3-bridge
      await fetch('/api/repid/sync', {
        method: 'POST',
        body: JSON.stringify(update),
      });
      await db.delete('pending-updates', update.agentId);
      synced++;
    } catch (e) {
      console.error('Sync failed for', update.agentId, e);
    }
  }
  
  return synced;
}

export async function cacheVirtue(
  virtueKey: string, 
  lang: string, 
  text: string
): Promise<void> {
  const db = await getDB();
  await db.put('localized-virtues', { virtueKey, lang, text });
}

export async function getVirtue(
  virtueKey: string, 
  lang: string
): Promise<string | null> {
  const db = await getDB();
  const virtue = await db.get('localized-virtues', `${virtueKey}-${lang}`);
  return virtue?.text ?? null;
}

function getTierFromScore(score: number): TrinityDB['repid-scores']['value']['tier'] {
  if (score >= 10000) return 'Canopy';
  if (score >= 7500) return 'Forest';
  if (score >= 5000) return 'Grove';
  if (score >= 1000) return 'Sapling';
  return 'Seedling';
}

// Auto-sync when coming online
if (typeof window !== 'undefined') {
  window.addEventListener('online', () => {
    syncPendingUpdates().then(count => {
      console.log(`Synced ${count} pending RepID updates`);
    });
  });
}

import { useState, useEffect } from 'react';

export type EnergyModeType = 'full' | 'balanced' | 'saver';

interface EnergyModeProps {
  onChange?: (mode: EnergyModeType) => void;
}

const MODE_CONFIG = {
  full: {
    label: 'Full Power',
    icon: '⚡',
    heartbeatInterval: 15000,  // 15 seconds
    description: 'Maximum features, higher battery usage',
  },
  balanced: {
    label: 'Balanced',
    icon: '⚖️',
    heartbeatInterval: 30000,  // 30 seconds
    description: 'Balanced performance and battery',
  },
  saver: {
    label: 'Saver',
    icon: '🌱',
    heartbeatInterval: 60000,  // 60 seconds
    description: 'Minimal battery, eco-friendly (RepID bonus)',
  },
};

export function EnergyMode({ onChange }: EnergyModeProps) {
  const [mode, setMode] = useState<EnergyModeType>('balanced');
  
  useEffect(() => {
    const saved = localStorage.getItem('energyMode') as EnergyModeType;
    if (saved && MODE_CONFIG[saved]) {
      setMode(saved);
    }
  }, []);
  
  const handleChange = (newMode: EnergyModeType) => {
    setMode(newMode);
    localStorage.setItem('energyMode', newMode);
    onChange?.(newMode);
  };
  
  return (
    <div className="flex flex-col gap-2 p-4 bg-gray-100 rounded-lg">
      <h3 className="font-semibold text-sm">Energy Mode</h3>
      <div className="flex gap-2">
        {(Object.keys(MODE_CONFIG) as EnergyModeType[]).map((m) => (
          <button
            key={m}
            onClick={() => handleChange(m)}
            className={`flex-1 p-2 rounded text-center transition-all ${
              mode === m

                : 'bg-white hover:bg-gray-200'

        </p>
      )}
    </div>
  );
}

export function getHeartbeatInterval(mode: EnergyModeType): number {
  return MODE_CONFIG[mode].heartbeatInterval;

'use client';

import { useState, useRef, useCallback } from 'react';

interface VoiceInputProps {
  onTranscript: (text: string, lang: string) => void;
  supportedLanguages?: string[];
}

export function VoiceInput({ 
  onTranscript, 
  supportedLanguages = ['en-US', 'es-ES', 'fr-FR'] 
}: VoiceInputProps) {
  const [isListening, setIsListening] = useState(false);
  const [currentLang, setCurrentLang] = useState(supportedLanguages[0]);
  const recognitionRef = useRef<SpeechRecognition | null>(null);
  
  const startListening = useCallback(() => {
    if (!('webkitSpeechRecognition' in window || 'SpeechRecognition' in window)) {
      alert('Speech recognition not supported in this browser');
      return;
    }
    
    const SpeechRecognition = window.SpeechRecognition || window.webkitSpeechRecognition;
    const recognition = new SpeechRecognition();
    
    recognition.lang = currentLang;
    recognition.continuous = false;
    recognition.interimResults = false;
    
    recognition.onresult = (event) => {
      const transcript = event.results[0][0].transcript;
      onTranscript(transcript, currentLang);
      setIsListening(false);
    };
    
    recognition.onerror = (event) => {
      console.error('Speech recognition error:', event.error);
      setIsListening(false);
    };
    
    recognition.onend = () => {
      setIsListening(false);
    };
    
    recognitionRef.current = recognition;
    recognition.start();
    setIsListening(true);
  }, [currentLang, onTranscript]);
  
  const stopListening = useCallback(() => {
    recognitionRef.current?.stop();
    setIsListening(false);
  }, []);
  
  return (
    <div className="flex flex-col gap-3 p-4">
      <select
        value={currentLang}
        onChange={(e) => setCurrentLang(e.target.value)}
        className="p-2 border rounded"
        disabled={isListening}
      >
        {supportedLanguages.map((lang) => (
          <option key={lang} value={lang}>
            {new Intl.DisplayNames([lang], { type: 'language' }).of(lang.split('-')[0])}
          </option>
        ))}
      </select>
      
      <button
        onClick={isListening ? stopListening : startListening}
        className={`p-4 rounded-full transition-all ${
          isListening 
            ? 'bg-red-500 animate-pulse' 
            : 'bg-blue-500 hover:bg-blue-600'
        } text-white`}
      >
        {isListening ? '🎙️ Listening...' : '🎤 Start Speaking'}
      </button>
    </div>
  );
}

pragma solidity ^0.8.20;

import "@openzeppelin/contracts/token/ERC721/ERC721.sol";
import "@openzeppelin/contracts/access/AccessControl.sol";
import "@openzeppelin/contracts/proxy/utils/UUPSUpgradeable.sol";
import "../governance/ConstitutionalGuard.sol";

/**
 * @title TrinityIdentityAdapter
 * @notice EIP-8004 compliant Identity Registry with ZKP extensions for Trinity
 * @dev Supports SBT (Soul Bound), DBT (Digital Bound), CBT (Charity Bound) tokens
 */
contract TrinityIdentityAdapter is ERC721, AccessControl, UUPSUpgradeable, ConstitutionalGuard {
    
    bytes32 public constant REGISTRAR_ROLE = keccak256("REGISTRAR_ROLE");
    bytes32 public constant JUDICIAL_ROLE = keccak256("JUDICIAL_ROLE");  // SHOFET
    
    enum IdentityType { SBT, DBT, CBT }
    
    struct Identity {
        IdentityType idType;
        bytes32 metadataHash;      // IPFS hash of metadata
        uint256 registeredAt;
        bool verified;
        bytes zkProof;             // ZKP proof for privacy
    }
    
    mapping(uint256 => Identity) public identities;
    mapping(address => uint256) public addressToTokenId;
    
    uint256 private _tokenIdCounter;
    
    // ZKP Verifier (Groth16)
    address public zkVerifier;
    
    event IdentityRegistered(
        uint256 indexed tokenId,
        address indexed owner,
        IdentityType idType,
        bytes32 metadataHash
    );
    
    event IdentityVerified(uint256 indexed tokenId, address verifier);
    
    constructor(address _zkVerifier) ERC721("Trinity Identity", "TRID") {
        _grantRole(DEFAULT_ADMIN_ROLE, msg.sender);
        _grantRole(REGISTRAR_ROLE, msg.sender);
        _grantRole(JUDICIAL_ROLE, msg.sender);
        zkVerifier = _zkVerifier;
    }
    
    /**
     * @notice Register a new DBT (Digital Bound Token) for an AI agent
     * @param owner The address that will own the agent identity
     * @param metadataHash IPFS hash of agent metadata
     * @param zkProof ZKP proof for privacy-preserving registration
     */
    function registerDBT(
        address owner,
        bytes32 metadataHash,
        bytes calldata zkProof
    ) external onlyRole(REGISTRAR_ROLE) returns (uint256) {
        require(addressToTokenId[owner] == 0, "Already registered");
        require(_verifyZKProof(zkProof), "Invalid ZKP");
        
        _tokenIdCounter++;
        uint256 tokenId = _tokenIdCounter;
        
        _safeMint(owner, tokenId);
        
        identities[tokenId] = Identity({
            idType: IdentityType.DBT,
            metadataHash: metadataHash,
            registeredAt: block.timestamp,
            verified: false,
            zkProof: zkProof
        });
        
        addressToTokenId[owner] = tokenId;
        
        emit IdentityRegistered(tokenId, owner, IdentityType.DBT, metadataHash);
        
        return tokenId;
    }
    
    /**
     * @notice Register a new SBT (Soul Bound Token) for a human with PoL
     * @param owner Human's verified address
     * @param metadataHash IPFS hash of identity metadata
     * @param polProof Proof of Life (4FA verification proof)
     */
    function registerSBT(
        address owner,
        bytes32 metadataHash,
        bytes calldata polProof
    ) external onlyRole(REGISTRAR_ROLE) returns (uint256) {
        require(addressToTokenId[owner] == 0, "Already registered");
        require(_verifyPoLProof(polProof), "Invalid PoL");
        
        _tokenIdCounter++;
        uint256 tokenId = _tokenIdCounter;
        
        _safeMint(owner, tokenId);
        
        identities[tokenId] = Identity({
            idType: IdentityType.SBT,
            metadataHash: metadataHash,
            registeredAt: block.timestamp,
            verified: true,  // PoL implies verification
            zkProof: polProof
        });
        
        addressToTokenId[owner] = tokenId;
        
        emit IdentityRegistered(tokenId, owner, IdentityType.SBT, metadataHash);
        
        return tokenId;
    }
    
    /**
     * @notice Register a CBT (Charity Bound Token) for verified nonprofits
     */
    function registerCBT(
        address charity,
        bytes32 metadataHash,
        bytes calldata verificationProof
    ) external onlyRole(REGISTRAR_ROLE) returns (uint256) {
        require(addressToTokenId[charity] == 0, "Already registered");
        
        _tokenIdCounter++;
        uint256 tokenId = _tokenIdCounter;
        
        _safeMint(charity, tokenId);
        
        identities[tokenId] = Identity({
            idType: IdentityType.CBT,
            metadataHash: metadataHash,
            registeredAt: block.timestamp,
            verified: false,  // Requires manual verification
            zkProof: verificationProof
        });
        
        addressToTokenId[charity] = tokenId;
        
        emit IdentityRegistered(tokenId, charity, IdentityType.CBT, metadataHash);
        
        return tokenId;
    }
    
    /**
     * @notice Verify an identity (requires JUDICIAL_ROLE - SHOFET)
     */
    function verifyIdentity(uint256 tokenId) 
        external 
        onlyRole(JUDICIAL_ROLE)
        actJustly  // Constitutional guard
    {
        require(_ownerOf(tokenId) != address(0), "Identity does not exist");
        identities[tokenId].verified = true;
        emit IdentityVerified(tokenId, msg.sender);
    }
    
    /**
     * @notice Get CAIP-10 formatted identity
     */
    function getCAIP10(uint256 tokenId) external view returns (string memory) {
        require(_ownerOf(tokenId) != address(0), "Identity does not exist");
        return string(abi.encodePacked(
            "eip155:",
            _toString(block.chainid),
            ":",
            _toHexString(address(this)),
            ":",
            _toString(tokenId)
        ));
    }
    
    // Soul-bound: prevent transfers
    function _update(
        address to,
        uint256 tokenId,
        address auth
    ) internal virtual override returns (address) {
        address from = _ownerOf(tokenId);
        
        // Allow minting (from == 0) but block transfers
        if (from != address(0) && to != address(0)) {
            revert("Soul-bound: transfers disabled");
        }
        
        return super._update(to, tokenId, auth);
    }
    
    function _verifyZKProof(bytes calldata proof) internal view returns (bool) {
        if (zkVerifier == address(0)) return true;  // Skip if no verifier set
        // TODO: Integrate Groth16 verifier call
        (bool success, bytes memory result) = zkVerifier.staticcall(
            abi.encodeWithSignature("verifyProof(bytes)", proof)
        );
        return success && abi.decode(result, (bool));
    }
    
    function _verifyPoLProof(bytes calldata proof) internal pure returns (bool) {
        // TODO: Implement 4FA PoL verification
        return proof.length > 0;
    }
    
    function _authorizeUpgrade(address newImplementation) 
        internal 
        override 
        onlyRole(JUDICIAL_ROLE) 
    {}
    
    function supportsInterface(bytes4 interfaceId)
        public
        view
        override(ERC721, AccessControl)
        returns (bool)
    {
        return super.supportsInterface(interfaceId);
    }
    
    // Helper functions
    function _toString(uint256 value) internal pure returns (string memory) {
        if (value == 0) return "0";
        uint256 temp = value;
        uint256 digits;
        while (temp != 0) {
            digits++;
            temp /= 10;
        }
        bytes memory buffer = new bytes(digits);
        while (value != 0) {
            digits--;
            buffer[digits] = bytes1(uint8(48 + uint256(value % 10)));
            value /= 10;
        }
        return string(buffer);
    }
    
    function _toHexString(address addr) internal pure returns (string memory) {
        bytes memory buffer = new bytes(42);
        buffer[0] = "0";
        buffer[1] = "x";
        bytes memory hexChars = "0123456789abcdef";
        for (uint256 i = 0; i < 20; i++) {
            buffer[2 + i * 2] = hexChars[uint8(uint160(addr) >> (8 * (19 - i)) >> 4) & 0xf];
            buffer[3 + i * 2] = hexChars[uint8(uint160(addr) >> (8 * (19 - i))) & 0xf];
        }
        return string(buffer);
    }
}

import "@openzeppelin/contracts/access/AccessControl.sol";

/**
 * @title RepIDRegistry
 * @notice Reputation scoring with Pedersen commitments for privacy
 * @dev Implements EIP-8004 Reputation Registry with Trinity extensions
 */
contract RepIDRegistry is AccessControl, ReentrancyGuard, ConstitutionalGuard {
    
    bytes32 public constant ORACLE_ROLE = keccak256("ORACLE_ROLE");
    bytes32 public constant VERITAS_ROLE = keccak256("VERITAS_ROLE");  // Truth verification
    
    uint256 public constant MAX_SCORE = 10000;
    uint256 public constant PEDERSEN_THRESHOLD = 500;  // Use commitments above this delta
    
    struct RepScore {
        uint256 score;
        uint256 lastUpdated;
        uint256 totalFeedback;
        bytes32 pedersenCommitment;  // For privacy when delta > threshold
    }
    
    struct Feedback {
        address from;
        int256 delta;
        bytes32 taskHash;
        uint256 timestamp;
        bool verified;
    }
    
    mapping(uint256 => RepScore) public scores;  // tokenId => RepScore
    mapping(uint256 => Feedback[]) public feedbackHistory;
    
    // Tier thresholds
    uint256 public constant TIER_SAPLING = 1000;
    uint256 public constant TIER_GROVE = 5000;
    uint256 public constant TIER_FOREST = 7500;
    uint256 public constant TIER_CANOPY = 10000;
    
    // Energy mode multipliers (1.0 = 100, 1.1 = 110)
    mapping(uint256 => uint8) public energyMultipliers;
    
    event ScoreUpdated(
        uint256 indexed tokenId,
        uint256 newScore,
        int256 delta,
        string tier
    );
    
    event FeedbackGiven(
        uint256 indexed toTokenId,
        address indexed from,
        int256 delta,
        bytes32 taskHash
    );
    
    event PedersenCommitmentStored(
        uint256 indexed tokenId,
        bytes32 commitment
    );
    
    constructor() {
        _grantRole(DEFAULT_ADMIN_ROLE, msg.sender);
        _grantRole(ORACLE_ROLE, msg.sender);
        _grantRole(VERITAS_ROLE, msg.sender);
    }
    
    /**
     * @notice Give feedback to an agent/user
     * @param tokenId The identity token receiving feedback
     * @param delta Score change (-100 to +100 per feedback)
     * @param taskHash Hash of the task this feedback relates to
     */
    function giveFeedback(
        uint256 tokenId,
        int256 delta,
        bytes32 taskHash
    ) external nonReentrant loveMercy {
        require(delta >= -100 && delta <= 100, "Delta out of range");
        
        RepScore storage rep = scores[tokenId];
        
        // Calculate new score with bounds
        int256 newScoreInt = int256(rep.score) + delta;
        if (newScoreInt < 0) newScoreInt = 0;
        if (newScoreInt > int256(MAX_SCORE)) newScoreInt = int256(MAX_SCORE);
        
        uint256 newScore = uint256(newScoreInt);
        
        // Apply energy multiplier if in saver mode
        if (energyMultipliers[tokenId] > 100 && delta > 0) {
            uint256 bonus = (uint256(delta) * energyMultipliers[tokenId]) / 100;
            newScore = newScore + bonus - uint256(delta);
            if (newScore > MAX_SCORE) newScore = MAX_SCORE;
        }
        
        // Store Pedersen commitment for large deltas (privacy)
        if (_abs(delta) > int256(PEDERSEN_THRESHOLD)) {
            bytes32 commitment = _computePedersenCommitment(tokenId, delta);
            rep.pedersenCommitment = commitment;
            emit PedersenCommitmentStored(tokenId, commitment);
        }
        
        rep.score = newScore;
        rep.lastUpdated = block.timestamp;
        rep.totalFeedback++;
        
        feedbackHistory[tokenId].push(Feedback({
            from: msg.sender,
            delta: delta,
            taskHash: taskHash,
            timestamp: block.timestamp,
            verified: false
        }));
        
        emit FeedbackGiven(tokenId, msg.sender, delta, taskHash);
        emit ScoreUpdated(tokenId, newScore, delta, _getTierName(newScore));
    }
    
    /**
     * @notice Verify feedback (VERITAS role)
     */
    function verifyFeedback(
        uint256 tokenId,
        uint256 feedbackIndex
    ) external onlyRole(VERITAS_ROLE) {
        require(feedbackIndex < feedbackHistory[tokenId].length, "Invalid index");
        feedbackHistory[tokenId][feedbackIndex].verified = true;
    }
    
    /**
     * @notice Set energy multiplier for eco-mode bonus
     */
    function setEnergyMultiplier(
        uint256 tokenId,
        uint8 multiplier  // 100 = 1.0x, 110 = 1.1x
    ) external onlyRole(ORACLE_ROLE) {
        require(multiplier >= 100 && multiplier <= 200, "Invalid multiplier");
        energyMultipliers[tokenId] = multiplier;
    }
    
    /**
     * @notice Get current tier for a token
     */
    function getTier(uint256 tokenId) external view returns (string memory) {
        return _getTierName(scores[tokenId].score);
    }
    
    /**
     * @notice Get unlocks for current tier
     */
    function getUnlocks(uint256 tokenId) external view returns (string[] memory) {
        uint256 score = scores[tokenId].score;
        
        if (score >= TIER_CANOPY) {
            string[] memory unlocks = new string[](5);
            unlocks[0] = "Full governance";
            unlocks[1] = "SHOFET candidacy";
            unlocks[2] = "Agent creation";
            unlocks[3] = "CBT recommendations";
            unlocks[4] = "DAO voting";
            return unlocks;
        }
        // ... add other tiers
        
        string[] memory basic = new string[](1);
        basic[0] = "Basic access";
        return basic;
    }
    
    function _getTierName(uint256 score) internal pure returns (string memory) {
        if (score >= TIER_CANOPY) return "Canopy";
        if (score >= TIER_FOREST) return "Forest";
        if (score >= TIER_GROVE) return "Grove";
        if (score >= TIER_SAPLING) return "Sapling";
        return "Seedling";
    }
    
    function _computePedersenCommitment(
        uint256 tokenId,
        int256 delta
    ) internal view returns (bytes32) {
        // Simplified Pedersen commitment (replace with circom in production)
        return keccak256(abi.encodePacked(
            tokenId,
            delta,
            block.timestamp,
            blockhash(block.number - 1)
        ));
    }
    
    function _abs(int256 x) internal pure returns (int256) {
        return x >= 0 ? x : -x;
    }
}

        bytes32 id;
        bytes32 actionHash;
        Chamber primaryChamber;
        bool executiveApproval;
        bool legislativeApproval;
        bool judicialApproval;
        bool executed;
        uint256 createdAt;
        uint256 expiresAt;
    }
    
    mapping(bytes32 => Proposal) public proposals;
    
    uint256 public constant PROPOSAL_DURATION = 7 days;
    
    event ProposalCreated(bytes32 indexed id, Chamber primaryChamber);
    event ChamberApproval(bytes32 indexed id, Chamber chamber, address approver);
    event ProposalExecuted(bytes32 indexed id);
    
    constructor() {
        _grantRole(DEFAULT_ADMIN_ROLE, msg.sender);
    }

     * @notice Create a proposal requiring dual-chamber approval
     */
    function createProposal(
        bytes32 actionHash,
        Chamber primaryChamber
    ) external returns (bytes32) {
        bytes32 id = keccak256(abi.encodePacked(
            actionHash,
            block.timestamp,
            msg.sender
        ));
        
        proposals[id] = Proposal({
            id: id,
            actionHash: actionHash,
            primaryChamber: primaryChamber,
            executiveApproval: false,
            legislativeApproval: false,
            judicialApproval: false,
            executed: false,
            createdAt: block.timestamp,
            expiresAt: block.timestamp + PROPOSAL_DURATION
        });
        
        emit ProposalCreated(id, primaryChamber);
        return id;
    }

     * @notice Approve proposal from Executive chamber (ORCH)
     */
    function approveAsExecutive(bytes32 proposalId) 
        external 
        onlyRole(EXECUTIVE_ROLE) 
    {
        _approve(proposalId, Chamber.Executive);
    }

     * @notice Approve proposal from Legislative chamber (GCM)

        onlyRole(LEGISLATIVE_ROLE)

    {
        _approve(proposalId, Chamber.Judicial);

        Proposal storage p = proposals[proposalId];
        require(p.id != bytes32(0), "Proposal not found");
        require(block.timestamp < p.expiresAt, "Proposal expired");
        require(!p.executed, "Already executed");
        
        if (chamber == Chamber.Executive) p.executiveApproval = true;
        if (chamber == Chamber.Legislative) p.legislativeApproval = true;
        if (chamber == Chamber.Judicial) p.judicialApproval = true;
        
        emit ChamberApproval(proposalId, chamber, msg.sender);
    }

    function hasRequiredApprovals(bytes32 proposalId) public view returns (bool) {

        uint8 approvalCount = 0;
        
        if (p.executiveApproval) approvalCount++;
        if (p.legislativeApproval) approvalCount++;
        if (p.judicialApproval) approvalCount++;
        
        return approvalCount >= 2;
    }

    config: CCIPConfig,
    privateKey: `0x${string}`
  ) {
    this.config = config;
    
    this.publicClient = createPublicClient({
      chain,
      transport: http(),
    });
    
    this.walletClient = createWalletClient({
      chain,
      transport: http(),
      account: privateKey as any, // Will derive account from key
    });
  }
  
  /**
   * Send a cross-chain message via CCIP
   */
  async sendMessage(
    message: CrossChainMessage,
    destinationAddress: `0x${string}`
  ): Promise<`0x${string}`> {
    const destinationSelector = CHAIN_SELECTORS[message.destinationChain];
    if (!destinationSelector) {
      throw new Error(`Unsupported destination chain: ${message.destinationChain}`);
    }
    
    const messageData = this.encodeMessage(message);
    
    const ccipMessage = {
      receiver: destinationAddress,
      data: messageData,
      tokenAmounts: [],
      feeToken: '0x0000000000000000000000000000000000000000' as `0x${string}`, // Pay in native
      extraArgs: '0x' as `0x${string}`,
    };
    
    // Get fee estimate
    const fee = await this.publicClient.readContract({
      address: this.config.sourceRouter,
      abi: CCIP_ROUTER_ABI,
      functionName: 'getFee',
      args: [destinationSelector, ccipMessage],
    });
    
    // Send with retry logic
    return this.sendWithRetry(
      destinationSelector,
      ccipMessage,
      fee as bigint
    );
  }
  
  private async sendWithRetry(
    destinationSelector: bigint,
    message: any,
    fee: bigint,
    attempt = 1
  ): Promise<`0x${string}`> {
    try {
      const hash = await this.walletClient.writeContract({
        address: this.config.sourceRouter,
        abi: CCIP_ROUTER_ABI,
        functionName: 'ccipSend',
        args: [destinationSelector, message],
        value: fee,
      });
      
      console.log(`CCIP message sent: ${hash}`);
      return hash;
      
    } catch (error) {
      if (attempt >= this.maxRetries) {
        throw new Error(`CCIP send failed after ${this.maxRetries} attempts: ${error}`);
      }
      
      const delay = this.retryDelayMs * Math.pow(2, attempt - 1); // Exponential backoff
      console.log(`Retry ${attempt}/${this.maxRetries} in ${delay}ms...`);
      
      await new Promise(resolve => setTimeout(resolve, delay));
      return this.sendWithRetry(destinationSelector, message, fee, attempt + 1);
    }
  }
  
  /**
   * Encode message for cross-chain transport
   */
  private encodeMessage(message: CrossChainMessage): `0x${string}` {
    const encoder = new TextEncoder();
    const jsonStr = JSON.stringify(message);
    const bytes = encoder.encode(jsonStr);
    return `0x${Buffer.from(bytes).toString('hex')}` as `0x${string}`;
  }
  
  /**
   * Estimate gas for cross-chain message
   */
  async estimateGas(
    message: CrossChainMessage,
    destinationAddress: `0x${string}`
  ): Promise<bigint> {
    const destinationSelector = CHAIN_SELECTORS[message.destinationChain];
    const messageData = this.encodeMessage(message);
    
    const ccipMessage = {
      receiver: destinationAddress,
      data: messageData,
      tokenAmounts: [],
      feeToken: '0x0000000000000000000000000000000000000000' as `0x${string}`,
      extraArgs: '0x' as `0x${string}`,
    };
    
    return await this.publicClient.readContract({
      address: this.config.sourceRouter,
      abi: CCIP_ROUTER_ABI,
      functionName: 'getFee',
      args: [destinationSelector, ccipMessage],
    }) as bigint;
  }
}

import { openDB, type IDBPDatabase } from 'idb';

interface KeyVaultDB {
  'api-keys': {
    key: string;
    value: {
      provider: string;
      encryptedKey: ArrayBuffer;
      iv: ArrayBuffer;
      createdAt: number;
      lastUsed: number;
    };
  };
}

/**
 * Client-side only key vault using WebCrypto + IndexedDB
 * Keys NEVER leave the browser
 */
export class BYOKVault {
  private db: IDBPDatabase<KeyVaultDB> | null = null;
  private masterKey: CryptoKey | null = null;
  
  async init(userPassword: string): Promise<void> {
    // Derive master key from password
    const encoder = new TextEncoder();
    const passwordBuffer = encoder.encode(userPassword);
    
    const keyMaterial = await crypto.subtle.importKey(
      'raw',
      passwordBuffer,
      'PBKDF2',
      false,
      ['deriveBits', 'deriveKey']
    );
    
    this.masterKey = await crypto.subtle.deriveKey(
      {
        name: 'PBKDF2',
        salt: encoder.encode('trinity-salt-v1'), // In production, use unique salt per user
        iterations: 100000,
        hash: 'SHA-256',
      },
      keyMaterial,
      { name: 'AES-GCM', length: 256 },
      false,
      ['encrypt', 'decrypt']
    );
    
    this.db = await openDB<KeyVaultDB>('byok-vault', 1, {
      upgrade(db) {
        db.createObjectStore('api-keys', { keyPath: 'provider' });
      },
    });
  }
  
  /**
   * Store an API key (encrypted client-side)
   */
  async storeKey(provider: string, apiKey: string): Promise<void> {
    if (!this.masterKey || !this.db) {
      throw new Error('Vault not initialized');
    }
    
    const encoder = new TextEncoder();
    const iv = crypto.getRandomValues(new Uint8Array(12));
    
    const encryptedKey = await crypto.subtle.encrypt(
      { name: 'AES-GCM', iv },
      this.masterKey,
      encoder.encode(apiKey)
    );
    
    await this.db.put('api-keys', {
      provider,
      encryptedKey,
      iv: iv.buffer,
      createdAt: Date.now(),
      lastUsed: Date.now(),
    });
  }
  
  /**
   * Retrieve and decrypt an API key
   */
  async getKey(provider: string): Promise<string | null> {
    if (!this.masterKey || !this.db) {
      throw new Error('Vault not initialized');
    }
    
    const record = await this.db.get('api-keys', provider);
    if (!record) return null;
    
    try {
      const decrypted = await crypto.subtle.decrypt(
        { name: 'AES-GCM', iv: new Uint8Array(record.iv) },
        this.masterKey,
        record.encryptedKey
      );
      
      // Update last used
      record.lastUsed = Date.now();
      await this.db.put('api-keys', record);
      
      const decoder = new TextDecoder();
      return decoder.decode(decrypted);
      
    } catch (error) {
      console.error('Decryption failed:', error);
      return null;
    }
  }
  
  /**
   * Check if a key exists without decrypting
   */
  async hasKey(provider: string): Promise<boolean> {
    if (!this.db) return false;
    const record = await this.db.get('api-keys', provider);
    return record !== undefined;
  }
  
  /**
   * Delete a stored key
   */
  async deleteKey(provider: string): Promise<void> {
    if (!this.db) return;
    await this.db.delete('api-keys', provider);
  }
  
  /**
   * List all stored providers (not the keys themselves)
   */
  async listProviders(): Promise<string[]> {
    if (!this.db) return [];
    const all = await this.db.getAll('api-keys');
    return all.map(r => r.provider);
  }
}

export type BYOKTier = 'none' | 'silver' | 'gold';

export interface BYOKStatus {
  tier: BYOKTier;
  hasKey: boolean;
  stakeAmount: number;  // HDG tokens staked
  multiplier: number;   // 1.0, 1.5, or 2.0
}

export interface MultiplierConfig {
  silverKeyOnly: number;      // 1.5x
  goldKeyAndStake: number;    // 2.0x
  minStakeForGold: number;    // 100 HDG
  ecoModeBonus: number;       // 1.1x additional
}

const DEFAULT_CONFIG: MultiplierConfig = {
  silverKeyOnly: 1.5,
  goldKeyAndStake: 2.0,
  minStakeForGold: 100,
  ecoModeBonus: 1.1,
};

export class MultiplierCalculator {
  private config: MultiplierConfig;
  
  constructor(config: Partial<MultiplierConfig> = {}) {
    this.config = { ...DEFAULT_CONFIG, ...config };
  }
  
  /**
   * Calculate RepID multiplier based on BYOK status
   */
  calculateMultiplier(status: BYOKStatus, ecoMode = false): number {
    let multiplier = 1.0;
    
    if (status.hasKey && status.stakeAmount >= this.config.minStakeForGold) {
      // Gold tier: BYOK + 100+ HDG stake
      multiplier = this.config.goldKeyAndStake;
    } else if (status.hasKey) {
      // Silver tier: BYOK only
      multiplier = this.config.silverKeyOnly;
    }
    
    // Apply eco mode bonus
    if (ecoMode) {
      multiplier *= this.config.ecoModeBonus;
    }
    
    return Math.round(multiplier * 100) / 100; // Round to 2 decimals
  }
  
  /**
   * Determine tier from status
   */
  getTier(status: BYOKStatus): BYOKTier {
    if (status.hasKey && status.stakeAmount >= this.config.minStakeForGold) {
      return 'gold';
    } else if (status.hasKey) {
      return 'silver';
    }
    return 'none';
  }
  
  /**
   * Calculate RepID points with multiplier applied
   */
  applyMultiplier(
    basePoints: number,
    status: BYOKStatus,
    ecoMode = false
  ): number {
    const multiplier = this.calculateMultiplier(status, ecoMode);
    return Math.floor(basePoints * multiplier);
  }
}

export const FOUNDATION_LLMS = [
  { provider: 'anthropic', model: 'claude-3-haiku-20240307', priority: 1 },
  { provider: 'groq', model: 'llama-3.1-8b-instant', priority: 2 },
  { provider: 'together', model: 'meta-llama/Llama-3-8b-chat-hf', priority: 3 },
];

export function getFallbackLLM(excludeProvider?: string) {
  const available = FOUNDATION_LLMS.filter(l => l.provider !== excludeProvider);
  return available.sort((a, b) => a.priority - b.priority)[0];
}
